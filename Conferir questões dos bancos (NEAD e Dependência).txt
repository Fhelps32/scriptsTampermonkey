// ==UserScript==
// @name         Conferir questões dos bancos (NEAD e Dependência)
// @namespace    https://nead.ugb.edu.br/
// @version      1.5
// @description  Onde usar: abra a categoria com a lista das salas e use o painel no canto da tela. Com um clique ele entra em todas as salas, abre os bancos de questões e procura questão duplicada, alternativa repetida e imagem quebrada. Questão ou alternativa que é só imagem é comparada pelo conteúdo da imagem, não pelo texto vazio.
// @author       Você
// @match        https://nead.ugb.edu.br/course/index.php*
// @match        https://dependencia.ugb.edu.br/course/index.php*
// @icon         https://nead.ugb.edu.br/favicon.ico
// @grant        none
// @run-at       document-idle
// ==/UserScript==

(function () {
  'use strict';

  // ----------------------------------------------------------------------
  // Ajustes
  // ----------------------------------------------------------------------
  const CONFIG = {
    // Quantas questões pedir por página da listagem do banco. O Moodle tem um
    // teto próprio; se vier menos, a paginação abaixo cobre o resto.
    questoesPorPagina: 1000,
    // Pausas, para não afogar o servidor numa varredura de dezenas de salas.
    pausaEntreSalasMs: 500,
    pausaEntreQuestoesMs: 120,
    pausaEntreImagensMs: 60,
    // Categorias do banco que não devem ser conferidas. A comparação é sem
    // acento e sem diferença de maiúscula, e vale se o nome CONTIVER o termo.
    // Ex.: [/DEFEITU/, /COM DEFEITO/] para pular as categorias onde vocês
    // guardam de propósito as questões com problema.
    ignorarCategorias: [],
    // Uma sala pode ter mais de um banco. Deixe vazio para conferir todos;
    // para conferir só um, ponha algo como /ADP[\s\-_.]*DEP/i.
    filtroDeBanco: null,
    // Páginas montadas por JavaScript: fallback por iframe, igual aos outros
    // scripts.
    usarIframe: true,
    esperaIframeMs: 12000,
    // Alternativa sem texto e sem imagem é relatada à parte, em vez de entrar
    // no bolo das "repetidas" — era daí que vinha boa parte do falso positivo.
    relatarAlternativaVazia: true,
    // Imagem é considerada quebrada quando o endereço não responde, responde
    // erro, vem vazia ou vem com um tipo que não é imagem (costuma ser uma
    // página de erro do Moodle disfarçada).
    exigirTipoDeImagem: true,
    // Ao terminar, baixa o relatório .txt com as questões que têm problema.
    baixarRelatorioAoFinal: true,
  };

  const ORIGEM = location.origin;
  const abs = (u) => new URL(u, ORIGEM).href;
  const dorme = (ms) => new Promise((r) => setTimeout(r, ms));
  const txt = (el) => (el && el.textContent || '').replace(/ /g, ' ').replace(/\s+/g, ' ').trim();

  // Comparação de texto: sem acento, sem diferença de maiúscula, espaços
  // colapsados. "Questão 1" e "QUESTAO  1" são a mesma coisa.
  const norm = (s) => String(s || '')
    .replace(/[​-‍﻿]/g, '')
    .normalize('NFKD').replace(/[̀-ͯ]/g, '')
    .toUpperCase().replace(/\s+/g, ' ').trim();

  // ----------------------------------------------------------------------
  // Rede
  // ----------------------------------------------------------------------
  async function baixaPagina(url) {
    try {
      const resp = await fetch(url, { credentials: 'same-origin' });
      const html = await resp.text();
      return { url, status: resp.status, ok: resp.ok, html, doc: new DOMParser().parseFromString(html, 'text/html') };
    } catch (e) {
      return { url, status: 0, ok: false, html: '', doc: null, erro: e.message };
    }
  }

  function carregaComJs(url, temOQuePreciso) {
    return new Promise((resolve) => {
      const frame = document.createElement('iframe');
      frame.style.cssText = 'position:fixed;left:-10000px;top:0;width:1280px;height:900px;border:0;';
      frame.src = url;
      document.body.appendChild(frame);
      const limite = Date.now() + CONFIG.esperaIframeMs;
      const timer = setInterval(() => {
        let doc = null;
        try { doc = frame.contentDocument; } catch (e) { /* ignora */ }
        if (doc && doc.readyState === 'complete' && temOQuePreciso(doc)) {
          clearInterval(timer);
          const copia = doc.cloneNode(true);
          setTimeout(() => frame.remove(), 100);
          resolve(copia);
          return;
        }
        if (Date.now() > limite) {
          clearInterval(timer);
          let ultimo = null;
          try { ultimo = frame.contentDocument ? frame.contentDocument.cloneNode(true) : null; } catch (e) { /* ignora */ }
          setTimeout(() => frame.remove(), 100);
          resolve(ultimo);
        }
      }, 400);
    });
  }

  // ----------------------------------------------------------------------
  // Salas da categoria aberta
  // ----------------------------------------------------------------------
  async function listarCursos() {
    const paginas = new Map([[location.href, document]]);
    document.querySelectorAll('.paging a[href], nav a[href*="page="]').forEach((a) => {
      if (!paginas.has(a.href)) paginas.set(a.href, null);
    });

    const cursos = new Map();
    for (const [url, cache] of paginas) {
      let doc = cache;
      if (!doc) { const r = await baixaPagina(url); if (!r.doc) continue; doc = r.doc; }
      const escopo = doc.querySelector('#region-main') || doc.body;
      escopo.querySelectorAll('a[href*="/course/view.php?id="]').forEach((a) => {
        const id = new URL(a.getAttribute('href'), ORIGEM).searchParams.get('id');
        const nome = txt(a);
        if (id && nome && !cursos.has(id)) cursos.set(id, nome);
      });
    }
    return [...cursos].map(([id, nome]) => ({ id, nome }));
  }

  const nomeCurto = (curso) => curso.nome.replace(/\[[^\]]*\]\s*/g, '').replace(/\s*[•⬩].*$/, '').trim();

  // Nos bancos de vocês o nome da questão costuma ser o enunciado inteiro.
  // Encurto para o relatório não virar uma parede de texto; o link vai junto.
  const nomeDaQuestao = (q) => {
    const n = (q.nome || '').trim();
    if (!n) return `questão ${q.id}`;
    return (n.length > 90 ? `${n.slice(0, 90)}…` : n) + ` (id ${q.id})`;
  };

  // ----------------------------------------------------------------------
  // Bancos de questões da sala
  // ----------------------------------------------------------------------
  // Mesma lógica do script de importar bancos: na versão nova do Moodle cada
  // banco é um módulo com seu próprio cmid.
  const RE_LINK_BANCO = /\/question\/edit\.php\?|\/mod\/qbank\/view\.php\?/;

  function linksDeBanco(doc) {
    if (!doc) return [];
    const vistos = new Set();
    return [...doc.querySelectorAll('a[href]')]
      .map((a) => {
        const href = a.getAttribute('href') || '';
        if (!RE_LINK_BANCO.test(href)) return null;
        const p = new URL(href, ORIGEM).searchParams;
        const cmid = p.get('cmid') || (href.includes('/mod/qbank/view.php') ? p.get('id') : null);
        if (!cmid || vistos.has(cmid)) return null;
        vistos.add(cmid);
        return { cmid, nome: txt(a) || `banco ${cmid}` };
      })
      .filter(Boolean);
  }

  async function bancosDoCurso(courseid) {
    const url = `${ORIGEM}/question/banks.php?courseid=${encodeURIComponent(courseid)}`;
    let r = await baixaPagina(url);
    let bancos = linksDeBanco(r.doc);
    if (!bancos.length && CONFIG.usarIframe) {
      const doc = await carregaComJs(url, (d) => linksDeBanco(d).length > 0);
      bancos = linksDeBanco(doc);
    }
    if (CONFIG.filtroDeBanco) bancos = bancos.filter((b) => CONFIG.filtroDeBanco.test(b.nome));
    return bancos;
  }

  // ----------------------------------------------------------------------
  // Categorias e questões de um banco
  // ----------------------------------------------------------------------
  // De onde vêm as categorias. Na versão nova do Moodle a página do banco NÃO
  // tem mais um <select> de categorias: o filtro é um componente montado por
  // JavaScript. O select de verdade, com "catid,contextid" em cada opção, está
  // nos formulários de exportar e de importar — é de lá que os outros scripts
  // leem, e é de lá que eu leio também. Ponho o de exportar na frente porque
  // esta varredura é só leitura.
  const PAGINAS_DE_CATEGORIA = [
    { rotulo: 'exportar', url: (cmid) => `${ORIGEM}/question/bank/exportquestions/export.php?cmid=${cmid}` },
    { rotulo: 'importar', url: (cmid) => `${ORIGEM}/question/bank/importquestions/import.php?cmid=${cmid}` },
    { rotulo: 'página do banco', url: (cmid) => `${ORIGEM}/question/edit.php?cmid=${cmid}` },
  ];

  // "Topo para X" (em inglês, "Top for X") é o nome que o Moodle dá à
  // categoria raiz do banco.
  const RE_CATEGORIA_TOPO = /^(TOPO PARA|TOP FOR)\b/;

  function categoriasDoDocumento(doc) {
    if (!doc) return [];
    const cats = new Map();
    const guarda = (valor, nome, contexto) => {
      const m = String(valor || '').trim().match(/^(\d+),(\d+)$/);
      if (!m || cats.has(m[0])) return;
      const limpo = String(nome || '').replace(/\s*\(\d+\)\s*$/, '').replace(/^[\s\u00a0]+/, '').trim();
      cats.set(m[0], { valor: m[0], id: m[1], contextid: m[2], nome: limpo || `categoria ${m[1]}`, contexto: contexto || '' });
    };

    // O select é um "selectgroups": as opções vêm agrupadas por contexto.
    doc.querySelectorAll('select[name="category"], select[name="cat"]').forEach((sel) => {
      sel.querySelectorAll('option').forEach((op) => {
        const grupo = op.closest('optgroup');
        guarda(op.getAttribute('value'), txt(op), grupo ? (grupo.getAttribute('label') || '') : '');
      });
    });

    // Último recurso: algum tema que ainda desenhe as categorias como links.
    if (!cats.size) {
      doc.querySelectorAll('a[href*="cat="]').forEach((a) => {
        const cat = new URL(a.getAttribute('href'), ORIGEM).searchParams.get('cat');
        guarda(cat, txt(a), '');
      });
    }
    return [...cats.values()];
  }

  // O contexto do banco. Preciso dele porque o select de categorias pode
  // trazer também as categorias dos contextos de cima (o curso, o sistema) —
  // e questão de outro banco entrando aqui viraria "duplicada" sem ser.
  function contextoDaPagina(html) {
    const m = String(html || '').match(/"contextid"\s*:\s*(\d+)/);
    return m ? m[1] : null;
  }

  async function categoriasDoBanco(cmid, anota) {
    for (const pagina of PAGINAS_DE_CATEGORIA) {
      const url = pagina.url(encodeURIComponent(cmid));
      let r = await baixaPagina(url);
      let cats = categoriasDoDocumento(r.doc);
      let html = r.html;

      if (!cats.length && CONFIG.usarIframe) {
        const doc = await carregaComJs(url, (d) => categoriasDoDocumento(d).length > 0);
        cats = categoriasDoDocumento(doc);
        if (doc) html = doc.documentElement ? doc.documentElement.outerHTML : html;
      }
      if (!cats.length) {
        if (anota) anota(`      sem categorias na tela de ${pagina.rotulo} (HTTP ${r.status})`);
        continue;
      }

      const contexto = contextoDaPagina(html);
      const doBanco = contexto ? cats.filter((c) => c.contextid === contexto) : cats;
      let usar = doBanco.length ? doBanco : cats;
      const deOutroContexto = cats.length - usar.length;

      // A categoria "topo" é só o guarda-chuva do banco: ela nunca tem questão
      // e aparece na tela de exportar ("Topo para …"). Tiro para não gastar
      // requisição com ela. Se por acaso ela for a única, mantenho.
      const semTopo = usar.filter((c) => !RE_CATEGORIA_TOPO.test(norm(c.nome)));
      if (semTopo.length) usar = semTopo;

      const ignoradas = [];
      if (CONFIG.ignorarCategorias.length) {
        usar = usar.filter((c) => {
          const pular = CONFIG.ignorarCategorias.some((re) => re.test(norm(c.nome)));
          if (pular) ignoradas.push(c.nome);
          return !pular;
        });
      }

      if (anota) {
        anota(`      categorias lidas na tela de ${pagina.rotulo}`
          + (contexto ? ` (contexto ${contexto})` : ' (não achei o contexto do banco)')
          + (deOutroContexto > 0 ? ` — ${deOutroContexto} de outro contexto ficaram de fora` : ''));
        if (ignoradas.length) anota(`      ignorando por configuração: ${ignoradas.join(' | ')}`);
      }
      return usar;
    }
    return [];
  }

  // O parâmetro "filter" é lido pelo Moodle com PARAM_RAW e passado por
  // json_decode (question/editlib.php). Ou seja, ele é uma STRING JSON — e não
  // campos aninhados filter[category][...], que era o que eu mandava antes.
  // Mandando array, o optional_param do Moodle reclama e o filtro chega sem
  // valores: a listagem vinha vazia e o banco parecia não ter questão nenhuma.
  function filtroDeCategoria(cat) {
    return JSON.stringify({
      category: {
        name: 'category',
        jointype: 1,
        values: [Number(cat.id)],
        filteroptions: { includesubcategories: false },
      },
    });
  }

  function urlDaListagem(cmid, cat, pagina, comFiltro) {
    const p = new URLSearchParams();
    p.set('cmid', String(cmid));
    p.set('cat', cat.valor);
    p.set('qpage', String(pagina));
    p.set('qperpage', String(CONFIG.questoesPorPagina));
    // Com o filtro eu fico independente da última preferência do usuário para
    // "incluir subcategorias". Sem ele, o Moodle decide sozinho — e aí a
    // varredura ainda funciona, porque as questões são juntadas por id e
    // repetição entre categorias se desfaz sozinha.
    if (comFiltro) p.set('filter', filtroDeCategoria(cat));
    return `${ORIGEM}/question/edit.php?${p.toString()}`;
  }

  // Cada linha da listagem tem a caixa de seleção "q<id>" e o link de
  // pré-visualização. Uso os dois, porque as colunas da listagem são
  // configuráveis e nem toda instalação mostra as mesmas.
  function questoesDoDocumento(doc) {
    if (!doc) return [];
    const achadas = new Map();

    const guarda = (id, nome) => {
      if (!id) return;
      const atual = achadas.get(id);
      if (!atual) achadas.set(id, { id, nome: nome || '' });
      else if (!atual.nome && nome) atual.nome = nome;
    };

    // O nome está no link que leva à edição daquela questão. Procuro por ele
    // antes de recorrer a palpites: pegar "o link mais comprido da linha"
    // trazia junto a coluna do texto da questão.
    const nomeDaLinha = (linha, id) => {
      if (!linha) return '';
      const doId = [...linha.querySelectorAll('a[href]')].find((a) => {
        const href = a.getAttribute('href') || '';
        if (/preview\.php/.test(href)) return false;
        const p = new URL(href, ORIGEM).searchParams;
        return p.get('id') === String(id) && /question\.php|editquestion/.test(href);
      });
      if (doId && txt(doId)) return txt(doId);
      const celula = linha.querySelector('.questionname, td.questionname');
      if (celula && txt(celula)) return txt(celula);
      const links = [...linha.querySelectorAll('a')].map((a) => txt(a)).filter((t) => t.length > 1);
      return links.sort((a, b) => b.length - a.length)[0] || '';
    };

    doc.querySelectorAll('input[type="checkbox"][name]').forEach((c) => {
      const m = (c.getAttribute('name') || '').match(/^q(\d+)$/);
      if (m) guarda(m[1], nomeDaLinha(c.closest('tr'), m[1]));
    });

    doc.querySelectorAll('a[href*="previewquestion/preview.php"], a[href*="/question/preview.php"]').forEach((a) => {
      const id = new URL(a.getAttribute('href'), ORIGEM).searchParams.get('id');
      guarda(id, nomeDaLinha(a.closest('tr'), id));
    });

    return [...achadas.values()];
  }

  // Duas formas de pedir a listagem: com e sem o filtro. O formato do filtro
  // mudou entre versões do Moodle, então não dependo de uma só. O iframe NÃO
  // entra aqui: ele custa segundos e só faz sentido quando o banco INTEIRO
  // vem vazio, o que é tratado em confereBanco — categoria vazia é normal.
  async function questoesDaCategoria(cmid, cat, comJs) {
    for (const comFiltro of [true, false]) {
      const questoes = new Map();
      for (let pagina = 0; pagina < 100; pagina++) {
        const url = urlDaListagem(cmid, cat, pagina, comFiltro);
        let achadas;
        if (comJs) {
          const doc = await carregaComJs(url, (d) => questoesDoDocumento(d).length > 0);
          achadas = questoesDoDocumento(doc);
        } else {
          const r = await baixaPagina(url);
          achadas = questoesDoDocumento(r.doc);
        }
        const novas = achadas.filter((q) => !questoes.has(q.id));
        novas.forEach((q) => questoes.set(q.id, q));
        if (!novas.length) break;
      }
      if (questoes.size) return { questoes: [...questoes.values()], comFiltro };
      if (comJs) break; // com JavaScript, uma passada basta
    }
    return { questoes: [], comFiltro: null };
  }

  // ----------------------------------------------------------------------
  // Imagens: uma leitura por endereço, reaproveitada em toda a varredura
  // ----------------------------------------------------------------------
  // O mesmo calendário aparece em dezenas de salas; ler uma vez e guardar o
  // resultado é o que faz essa varredura caber em alguns minutos.
  const cacheDeImagens = new Map();

  // Impressão digital do conteúdo da imagem. Prefiro SHA-256 do navegador;
  // onde ele não estiver disponível (contexto sem HTTPS, por exemplo), caio
  // numa conta simples feita aqui mesmo. O importante é que uma falha ao
  // calcular a impressão NUNCA faça a imagem ser acusada de quebrada: ler o
  // arquivo e identificá-lo são duas coisas separadas.
  async function impressaoDoConteudo(buffer) {
    try {
      if (crypto && crypto.subtle && crypto.subtle.digest) {
        const bytes = await crypto.subtle.digest('SHA-256', buffer);
        return [...new Uint8Array(bytes)].map((b) => b.toString(16).padStart(2, '0')).join('');
      }
    } catch (e) { /* cai no cálculo abaixo */ }
    return impressaoSimples(buffer);
  }

  // FNV-1a em dois passes (um do começo, outro de trás para frente) mais o
  // tamanho. Não é criptografia, mas distingue imagens diferentes de sobra.
  function impressaoSimples(buffer) {
    const b = new Uint8Array(buffer);
    let h1 = 0x811c9dc5, h2 = 0x01000193;
    for (let i = 0; i < b.length; i++) {
      h1 = ((h1 ^ b[i]) * 0x01000193) >>> 0;
      h2 = ((h2 ^ b[b.length - 1 - i]) * 0x811c9dc5) >>> 0;
    }
    const hex = (n) => n.toString(16).padStart(8, '0');
    return `simples${hex(h1)}${hex(h2)}${hex(b.length)}`;
  }

  async function leImagem(url) {
    if (cacheDeImagens.has(url)) return cacheDeImagens.get(url);
    const promessa = (async () => {
      try {
        const resp = await fetch(url, { credentials: 'same-origin' });
        if (!resp.ok) return { ok: false, motivo: `o endereço respondeu HTTP ${resp.status}` };
        const tipo = (resp.headers.get('content-type') || '').toLowerCase();
        const buffer = await resp.arrayBuffer();
        if (!buffer.byteLength) return { ok: false, motivo: 'o arquivo da imagem veio vazio' };
        // Uma página de erro do Moodle responde 200 com text/html — do ponto
        // de vista do aluno, a imagem está quebrada do mesmo jeito.
        if (CONFIG.exigirTipoDeImagem && tipo && !tipo.startsWith('image/')) {
          return { ok: false, motivo: `o endereço devolveu "${tipo.split(';')[0]}" em vez de uma imagem` };
        }
        // Daqui para baixo a imagem existe e foi lida: ela não está quebrada,
        // aconteça o que acontecer no cálculo da impressão digital.
        return { ok: true, hash: await impressaoDoConteudo(buffer), bytes: buffer.byteLength };
      } catch (e) {
        return { ok: false, motivo: `não consegui abrir (${e.message})` };
      }
    })();
    cacheDeImagens.set(url, promessa);
    return promessa;
  }

  // ----------------------------------------------------------------------
  // Leitura de uma questão
  // ----------------------------------------------------------------------
  function urlDaPreVisualizacao(cmid, questaoId) {
    return `${ORIGEM}/question/bank/previewquestion/preview.php?id=${encodeURIComponent(questaoId)}`
      + `&cmid=${encodeURIComponent(cmid)}`;
  }

  // Transforma o conteúdo de um pedaço da questão em texto comparável. Cada
  // imagem entra como uma marca do seu CONTEÚDO, não do seu endereço:
  //   - pelo endereço, duas cópias da mesma imagem em questões diferentes
  //     pareceriam diferentes (o endereço carrega o id da questão);
  //   - pelo texto, duas imagens diferentes pareceriam iguais, porque o texto
  //     dos dois lados é vazio. Era esse o falso positivo.
  async function conteudoComparavel(elemento, registraImagemQuebrada) {
    if (!elemento) return { texto: '', temImagem: false, vazio: true };

    const copia = elemento.cloneNode(true);
    let temImagem = false;

    for (const img of [...copia.querySelectorAll('img')]) {
      const bruto = img.getAttribute('src') || '';
      // Imagem decorativa do editor (emoticon, ícone) não deveria pesar na
      // comparação, mas ela também aparece para o aluno, então mantenho.
      if (!bruto) { img.replaceWith(copia.ownerDocument.createTextNode(' [img-sem-endereco] ')); temImagem = true; continue; }
      const endereco = abs(bruto);
      const leitura = await leImagem(endereco);
      temImagem = true;
      if (leitura.ok) {
        img.replaceWith(copia.ownerDocument.createTextNode(` [img#${leitura.hash.slice(0, 16)}] `));
      } else {
        if (registraImagemQuebrada) registraImagemQuebrada(endereco, leitura.motivo);
        img.replaceWith(copia.ownerDocument.createTextNode(` [img-quebrada:${endereco}] `));
      }
      await dorme(CONFIG.pausaEntreImagensMs);
    }

    // Vídeo e áudio incorporados contam pelo endereço: não há como ler o
    // conteúdo deles aqui e dois endereços diferentes são conteúdos diferentes.
    copia.querySelectorAll('video[src], audio[src], source[src], iframe[src], embed[src], object[data]').forEach((el) => {
      const endereco = el.getAttribute('src') || el.getAttribute('data') || '';
      el.replaceWith(copia.ownerDocument.createTextNode(` [midia:${abs(endereco)}] `));
      temImagem = true;
    });

    const texto = norm(txt(copia));
    return { texto, temImagem, vazio: !texto };
  }

  async function leQuestao(cmid, questao, anota) {
    const url = urlDaPreVisualizacao(cmid, questao.id);
    let r = await baixaPagina(url);
    let formulacao = r.doc && r.doc.querySelector('.formulation');
    if (!formulacao && CONFIG.usarIframe) {
      const doc = await carregaComJs(url, (d) => !!d.querySelector('.formulation'));
      formulacao = doc && doc.querySelector('.formulation');
    }
    if (!formulacao) throw new Error(`não consegui abrir a pré-visualização (HTTP ${r.status})`);

    const quebradas = [];
    const registra = (endereco, motivo) => {
      if (!quebradas.some((q) => q.endereco === endereco)) quebradas.push({ endereco, motivo });
    };

    const enunciado = await conteudoComparavel(formulacao.querySelector('.qtext'), registra);

    // O Moodle desenha cada alternativa assim (renderer do multichoice):
    //   <input> <div class="d-flex w-auto"><span class="answernumber">a. </span>
    //            <div class="flex-fill ms-1">texto/imagem da alternativa</div></div>
    // Ou seja, a letra fica FORA do .flex-fill — o conteúdo vem limpo. As
    // outras duas tentativas servem para tipos de questão que não usam essa
    // estrutura.
    let caixas = [...formulacao.querySelectorAll('.answer .flex-fill')];
    if (!caixas.length) caixas = [...formulacao.querySelectorAll('.answer [data-region="answer-label"]')];
    if (!caixas.length) caixas = [...formulacao.querySelectorAll('.answer .r0, .answer .r1')];

    const alternativas = [];
    for (const [i, caixa] of caixas.entries()) {
      const c = await conteudoComparavel(caixa, registra);
      alternativas.push({ indice: i + 1, letra: letraDaAlternativa(caixa, i), ...c });
    }

    // Imagem que esteja fora do enunciado e das alternativas (num feedback,
    // por exemplo) também é imagem quebrada para quem abre a questão.
    const jaVistas = new Set();
    formulacao.querySelectorAll('.qtext img, .answer img').forEach((img) => jaVistas.add(img.getAttribute('src') || ''));
    for (const img of [...formulacao.querySelectorAll('img')]) {
      const bruto = img.getAttribute('src') || '';
      if (!bruto || jaVistas.has(bruto)) continue;
      const leitura = await leImagem(abs(bruto));
      if (!leitura.ok) registra(abs(bruto), leitura.motivo);
      await dorme(CONFIG.pausaEntreImagensMs);
    }

    return { enunciado, alternativas, imagensQuebradas: quebradas, url };
  }

  // A letra ("a.", "b.") está no irmão .answernumber, então subo um nível
  // para achá-la sem trazê-la para dentro da comparação.
  function letraDaAlternativa(caixa, i) {
    const candidatos = [caixa.parentElement, caixa.closest('[data-region="answer-label"]'), caixa.closest('.r0, .r1')];
    for (const pai of candidatos) {
      const marca = pai && pai.querySelector('.answernumber');
      const t = txt(marca).replace(/[.)\s]+$/, '');
      if (t) return t;
    }
    return String.fromCharCode(97 + i);
  }

  // ----------------------------------------------------------------------
  // As três regras
  // ----------------------------------------------------------------------
  function assinaturaDaQuestao(leitura) {
    const alts = leitura.alternativas.map((a) => a.texto).sort();
    return JSON.stringify([leitura.enunciado.texto, alts]);
  }

  function alternativasRepetidas(leitura) {
    const mapa = new Map();
    leitura.alternativas.forEach((a) => {
      // Alternativa sem nada (sem texto e sem imagem) sai da comparação: era
      // ela que fazia três alternativas em branco virarem "repetidas".
      if (a.vazio && !a.temImagem) return;
      if (!mapa.has(a.texto)) mapa.set(a.texto, []);
      mapa.get(a.texto).push(a);
    });
    const repetidas = [];
    mapa.forEach((grupo, texto) => {
      if (grupo.length > 1) {
        repetidas.push({
          texto: resumoDeTexto(texto),
          quantidade: grupo.length,
          letras: grupo.map((a) => a.letra).join(', '),
          soImagem: grupo.every((a) => a.temImagem && !temTextoAlemDaMarca(a.texto)),
        });
      }
    });
    return repetidas;
  }

  function alternativasVazias(leitura) {
    return leitura.alternativas.filter((a) => a.vazio && !a.temImagem).map((a) => a.letra);
  }

  // O texto comparado já passou pelo norm(), que deixa tudo em maiúsculas,
  // então o regex precisa ignorar a caixa — sem o "i" a marca não casava e a
  // alternativa que é só imagem era descrita como se tivesse texto.
  const RE_MARCAS = /\[(?:img#[0-9a-f]+|img-quebrada:[^\]]*|img-sem-endereco|midia:[^\]]*)\]/gi;
  const temTextoAlemDaMarca = (t) => !!norm(String(t).replace(RE_MARCAS, ' '));

  function resumoDeTexto(t) {
    const limpo = String(t).replace(/\[img#([0-9a-f]{6})[0-9a-f]*\]/gi, '[imagem $1]');
    return limpo.length > 120 ? `${limpo.slice(0, 120)}…` : limpo;
  }

  // ----------------------------------------------------------------------
  // Varredura de um banco
  // ----------------------------------------------------------------------
  async function confereBanco(curso, banco, anota) {
    const resultado = {
      curso, banco, questoes: 0, lidas: 0,
      duplicadas: [], repetidas: [], vazias: [], imagens: [], falhas: [], naoLidas: [],
    };

    const cats = await categoriasDoBanco(banco.cmid, anota);
    if (!cats.length) {
      resultado.falhas.push('não achei as categorias deste banco');
      return resultado;
    }
    anota(`      ${cats.length} categoria(s): ${cats.map((c) => c.nome).join(' | ')}`);

    const questoes = new Map();
    const juntar = async (comJs) => {
      for (const cat of cats) {
        if (interromper) break;
        const r = await questoesDaCategoria(banco.cmid, cat, comJs);
        if (r.comFiltro === false) anota(`      categoria "${cat.nome}": li a listagem sem o filtro`);
        r.questoes.forEach((q) => {
          if (!questoes.has(q.id)) questoes.set(q.id, { ...q, categoria: cat.nome, catValor: cat.valor });
        });
      }
    };

    await juntar(false);
    if (!questoes.size && CONFIG.usarIframe && !interromper) {
      anota('      nenhuma questão pelo caminho normal — tentando com JavaScript');
      await juntar(true);
    }
    resultado.questoes = questoes.size;
    anota(`      ${questoes.size} questão(ões) no banco`);

    const porAssinatura = new Map();

    for (const q of questoes.values()) {
      if (interromper) break;
      try {
        const leitura = await leQuestao(banco.cmid, q, anota);
        resultado.lidas += 1;

        const assinatura = assinaturaDaQuestao(leitura);
        if (!porAssinatura.has(assinatura)) porAssinatura.set(assinatura, []);
        porAssinatura.get(assinatura).push({ ...q, url: leitura.url });

        alternativasRepetidas(leitura).forEach((rep) => {
          resultado.repetidas.push({ questao: q, ...rep, url: leitura.url });
        });

        if (CONFIG.relatarAlternativaVazia) {
          const vazias = alternativasVazias(leitura);
          if (vazias.length) resultado.vazias.push({ questao: q, letras: vazias.join(', '), url: leitura.url });
        }

        leitura.imagensQuebradas.forEach((img) => {
          resultado.imagens.push({ questao: q, ...img, url: leitura.url });
        });
      } catch (e) {
        resultado.falhas.push(`questão ${q.id}${q.nome ? ` ("${q.nome.slice(0, 60)}")` : ''}: ${e.message}`);
        resultado.naoLidas.push({ questao: q, motivo: e.message });
      }
      await dorme(CONFIG.pausaEntreQuestoesMs);
    }

    porAssinatura.forEach((grupo) => {
      if (grupo.length > 1) resultado.duplicadas.push(grupo);
    });

    return resultado;
  }

  // ----------------------------------------------------------------------
  // Execução
  // ----------------------------------------------------------------------
  let rodando = false, interromper = false;
  let relatorio = [];
  let resumo = '';

  async function verificar() {
    if (rodando) return;
    rodando = true; interromper = false; relatorio = [];
    travaBotoes(true); botaoParar.style.display = ''; botaoCopiar.style.display = 'none';
    botaoTxt.style.display = 'none';
    painelLog.innerHTML = '';

    try {
      const cursos = await listarCursos();
      log(`${cursos.length} sala(s) nesta categoria.`);

      for (const curso of cursos) {
        if (interromper) { log('Interrompido por você.', 'aviso'); break; }
        const curto = nomeCurto(curso);
        log(curto);

        let bancos = [];
        try {
          bancos = await bancosDoCurso(curso.id);
        } catch (e) {
          log(`   erro ao procurar os bancos: ${e.message}`, 'erro');
          relatorio.push({ curso, banco: null, falhas: [`erro ao procurar os bancos: ${e.message}`],
            questoes: 0, lidas: 0, duplicadas: [], repetidas: [], vazias: [], imagens: [] });
          continue;
        }

        if (!bancos.length) {
          log('   sem banco de questões', 'aviso');
          relatorio.push({ curso, banco: null, semBanco: true, falhas: [],
            questoes: 0, lidas: 0, duplicadas: [], repetidas: [], vazias: [], imagens: [] });
          await dorme(CONFIG.pausaEntreSalasMs);
          continue;
        }

        for (const banco of bancos) {
          if (interromper) break;
          log(`   Banco: ${banco.nome}`);
          const r = await confereBanco(curso, banco, (m) => log(m));
          relatorio.push(r);
          log(`      ${linhaDoBanco(r)}`, corDoBanco(r));
        }
        await dorme(CONFIG.pausaEntreSalasMs);
      }

      resumo = textoResumo();
      log('');
      resumo.split('\n').forEach((l) => log(l));
      if (relatorio.length) {
        botaoCopiar.style.display = '';
        if (txtDoRelatorio().quantidade) {
          botaoTxt.style.display = '';
          if (CONFIG.baixarRelatorioAoFinal) baixarTxt();
        }
      }
    } catch (e) {
      log(`Parou: ${e.message}`, 'erro');
    } finally {
      rodando = false; interromper = false;
      travaBotoes(false); botaoParar.style.display = 'none'; botaoParar.textContent = 'Parar';
    }
  }

  const totalDeProblemas = (r) => r.duplicadas.length + r.repetidas.length + r.imagens.length;

  function linhaDoBanco(r) {
    if (r.falhas.length && !r.lidas) return `não consegui conferir: ${r.falhas[0]}`;
    const partes = [];
    partes.push(`${r.lidas}/${r.questoes} questão(ões) lidas`);
    partes.push(r.duplicadas.length ? `${r.duplicadas.length} grupo(s) de duplicadas` : 'sem duplicadas');
    partes.push(r.repetidas.length ? `${r.repetidas.length} alternativa(s) repetida(s)` : 'sem alternativa repetida');
    partes.push(r.imagens.length ? `${r.imagens.length} imagem(ns) quebrada(s)` : 'sem imagem quebrada');
    if (r.vazias.length) partes.push(`${r.vazias.length} questão(ões) com alternativa em branco`);
    if (r.falhas.length) partes.push(`${r.falhas.length} questão(ões) não lida(s)`);
    return partes.join(' · ');
  }

  const corDoBanco = (r) => (totalDeProblemas(r) ? 'erro' : (r.falhas.length ? 'aviso' : 'ok'));

  function textoResumo() {
    const linhas = [];
    const bancos = relatorio.filter((r) => r.banco);
    const semBanco = relatorio.filter((r) => r.semBanco);
    const comProblema = bancos.filter((r) => totalDeProblemas(r) > 0);

    linhas.push('════════ CONFERÊNCIA DAS QUESTÕES DOS BANCOS ════════');
    linhas.push(`Salas: ${new Set(relatorio.map((r) => r.curso.id)).size}`);
    linhas.push(`Bancos conferidos: ${bancos.length}`);
    linhas.push(`Questões lidas: ${bancos.reduce((s, r) => s + r.lidas, 0)} de ${bancos.reduce((s, r) => s + r.questoes, 0)}`);
    linhas.push(`Grupos de questões duplicadas: ${bancos.reduce((s, r) => s + r.duplicadas.length, 0)}`);
    linhas.push(`Alternativas repetidas: ${bancos.reduce((s, r) => s + r.repetidas.length, 0)}`);
    linhas.push(`Imagens quebradas: ${bancos.reduce((s, r) => s + r.imagens.length, 0)}`);
    linhas.push(`Questões com alternativa em branco: ${bancos.reduce((s, r) => s + r.vazias.length, 0)}`);

    if (!comProblema.length) {
      linhas.push('');
      linhas.push('Nenhum problema encontrado nos bancos conferidos.');
    }

    comProblema.forEach((r) => {
      linhas.push('');
      linhas.push(`── ${nomeCurto(r.curso)} · ${r.banco.nome}`);

      r.duplicadas.forEach((grupo, i) => {
        linhas.push(`   QUESTÕES DUPLICADAS, grupo ${i + 1} (${grupo.length}):`);
        grupo.forEach((q) => linhas.push(`     - [${q.categoria}] ${nomeDaQuestao(q)} — ${q.url}`));
      });

      r.repetidas.forEach((rep) => {
        const so = rep.soImagem ? ' (as duas são a MESMA imagem)' : '';
        linhas.push(`   ALTERNATIVA REPETIDA em "${nomeDaQuestao(rep.questao)}" — letras ${rep.letras} (${rep.quantidade}x)${so}`);
        linhas.push(`     "${rep.texto}"`);
        linhas.push(`     ${rep.url}`);
      });

      r.imagens.forEach((img) => {
        linhas.push(`   IMAGEM QUEBRADA em "${nomeDaQuestao(img.questao)}": ${img.motivo}`);
        linhas.push(`     ${img.endereco}`);
      });

      r.vazias.forEach((v) => {
        linhas.push(`   ALTERNATIVA EM BRANCO em "${nomeDaQuestao(v.questao)}" — letras ${v.letras}`);
      });

      if (r.falhas.length) {
        linhas.push(`   NÃO CONSEGUI LER (${r.falhas.length}):`);
        r.falhas.forEach((f) => linhas.push(`     - ${f}`));
      }
    });

    if (semBanco.length) {
      linhas.push('');
      linhas.push(`SALAS SEM BANCO DE QUESTÕES (${semBanco.length}):`);
      semBanco.forEach((r) => linhas.push(`  - ${nomeCurto(r.curso)}`));
    }

    return linhas.join('\n');
  }

  function copiarResumo() {
    navigator.clipboard.writeText(resumo)
      .then(() => log('Resumo copiado.', 'ok'))
      .catch(() => log('Não consegui copiar. Selecione o texto do log e copie à mão.', 'erro'));
  }

  // ----------------------------------------------------------------------
  // Relatório .txt
  // ----------------------------------------------------------------------
  // Só o que serve para agir: o problema, a categoria (AV1 / AV2) e o link.
  //
  // O link é a listagem do banco, na categoria da questão, com ela destacada.
  // Não existe um link que mova a questão sozinho: no Moodle 5 a categoria
  // fica travada na tela de editar ("Categoria atual", congelada — o campo
  // "Salvar em categoria" só aparece ao duplicar) e mover virou uma janela
  // dentro da listagem ("Mover para…"). Então o link leva até a questão e o
  // parâmetro lastchanged faz o Moodle pintá-la de amarelo (classe
  // "highlight", em question/classes/local/bank/view.php).
  function linkParaMover(r, q) {
    const p = new URLSearchParams();
    p.set('cmid', String(r.banco.cmid));
    if (q.catValor) p.set('cat', q.catValor);
    // Página grande o bastante para a questão não cair numa página 2.
    p.set('qperpage', String(CONFIG.questoesPorPagina));
    p.set('lastchanged', String(q.id));
    return `${ORIGEM}/question/edit.php?${p.toString()}`;
  }

  // Uma entrada por QUESTÃO, mesmo que ela tenha mais de um problema: é um
  // link só, e você move a questão uma vez só.
  function questoesComProblema() {
    const porSala = new Map();

    const anota = (r, q, problema) => {
      const sala = nomeCurto(r.curso);
      if (!porSala.has(sala)) porSala.set(sala, new Map());
      const daSala = porSala.get(sala);
      const chave = `${r.banco.cmid}:${q.id}`;
      if (!daSala.has(chave)) {
        daSala.set(chave, { id: Number(q.id), categoria: q.categoria || '', link: linkParaMover(r, q), problemas: new Map() });
      }
      const item = daSala.get(chave);
      item.problemas.set(problema, (item.problemas.get(problema) || 0) + 1);
    };

    relatorio.forEach((r) => {
      if (!r.banco) return;
      r.duplicadas.forEach((grupo) => grupo.forEach((q, i) => {
        anota(r, q, `Questão duplicada (cópia ${i + 1} de ${grupo.length})`);
      }));
      r.repetidas.forEach((rep) => {
        anota(r, rep.questao, `Alternativa repetida (letras ${rep.letras})${rep.soImagem ? ' — mesma imagem' : ''}`);
      });
      r.imagens.forEach((img) => anota(r, img.questao, 'Imagem quebrada'));
      r.vazias.forEach((v) => anota(r, v.questao, `Alternativa em branco (letras ${v.letras})`));
      (r.naoLidas || []).forEach((n) => anota(r, n.questao, 'Não consegui conferir esta questão'));
    });

    return porSala;
  }

  function txtDoRelatorio() {
    const porSala = questoesComProblema();
    let quantidade = 0;
    const blocos = [];

    porSala.forEach((itens, sala) => {
      // AV1 antes de AV2, e dentro de cada uma na ordem do banco.
      const lista = [...itens.values()].sort((a, b) =>
        a.categoria.localeCompare(b.categoria, 'pt-BR', { numeric: true }) || a.id - b.id);
      quantidade += lista.length;

      const linhas = ['==================================================', sala, ''];
      lista.forEach((item) => {
        const problemas = [...item.problemas].map(([p, n]) => (n > 1 ? `${p} (${n}x)` : p)).join(' · ');
        linhas.push(`[${item.categoria || 'sem categoria'}] ${problemas}`);
        linhas.push(item.link);
        linhas.push('');
      });
      blocos.push(linhas.join('\r\n'));
    });

    // Banco que nem deu para abrir não tem questão para listar, mas precisa
    // aparecer: senão o relatório dá a entender que ele foi conferido.
    const naoConferidos = relatorio
      .filter((r) => r.banco && !r.lidas && r.falhas.some((f) => !/^questão \d+/.test(f)))
      .map((r) => `${nomeCurto(r.curso)} — ${r.falhas.find((f) => !/^questão \d+/.test(f))}`);

    const agora = new Date();
    const data = agora.toLocaleDateString('pt-BR') + ' ' + agora.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' });
    const salas = new Set(relatorio.map((r) => r.curso.id)).size;

    const cabecalho = [
      'CONFERÊNCIA DAS QUESTÕES DOS BANCOS',
      `${data} · ${salas} sala(s) conferida(s) · ${quantidade} questão(ões) com problema`,
      '',
      'Cada link abre o banco na categoria da questão, com ela destacada em amarelo.',
      'Marque a caixinha dela e use "Mover para…" para trocá-la de categoria.',
      '',
    ];

    const rodape = naoConferidos.length
      ? ['==================================================', 'BANCOS QUE NÃO CONSEGUI CONFERIR', '',
        ...naoConferidos, '']
      : [];

    return {
      quantidade: quantidade + naoConferidos.length,
      // BOM e CRLF: o Bloco de Notas do Windows abre com os acentos certos.
      texto: '\ufeff' + [...cabecalho, ...blocos, ...rodape].join('\r\n'),
    };
  }

  function baixarTxt() {
    const relatorioTxt = txtDoRelatorio();
    // Relatório vazio não ajuda ninguém: digo no log e não baixo.
    if (!relatorioTxt.quantidade) {
      log('Nenhum problema encontrado — não há nada para pôr no relatório.', 'ok');
      return;
    }
    const blob = new Blob([relatorioTxt.texto], { type: 'text/plain;charset=utf-8' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    const agora = new Date().toISOString().slice(0, 16).replace(/[:T]/g, '-');
    a.download = `conferencia-questoes-${agora}.txt`;
    document.body.appendChild(a);
    a.click();
    setTimeout(() => { URL.revokeObjectURL(a.href); a.remove(); }, 1000);
    log(`Relatório gerado com ${relatorioTxt.quantidade} questão(ões) com problema.`, 'ok');
  }

  // ----------------------------------------------------------------------
  // Diagnóstico
  // ----------------------------------------------------------------------
  // Olha UMA sala e conta tudo que encontrou em cada etapa. Serve para quando
  // a varredura não acha questão: em vez de adivinhar, o log mostra o que a
  // página devolveu de fato.
  async function diagnosticar() {
    if (rodando) return;
    rodando = true; interromper = false;
    travaBotoes(true); botaoParar.style.display = '';
    painelLog.innerHTML = '';

    try {
      const cursos = await listarCursos();
      log(`${cursos.length} sala(s) nesta categoria. Vou diagnosticar a primeira que tiver banco.`);

      for (const curso of cursos) {
        if (interromper) break;
        log(`Sala: ${nomeCurto(curso)} (id ${curso.id})`);

        const bancos = await bancosDoCurso(curso.id);
        log(`   bancos achados: ${bancos.length}${bancos.length ? ` — ${bancos.map((b) => `"${b.nome}" (cmid ${b.cmid})`).join(', ')}` : ''}`);
        if (!bancos.length) continue;

        const banco = bancos[0];

        for (const pagina of PAGINAS_DE_CATEGORIA) {
          const url = pagina.url(encodeURIComponent(banco.cmid));
          const r = await baixaPagina(url);
          const selects = r.doc ? r.doc.querySelectorAll('select[name="category"], select[name="cat"]').length : 0;
          const cats = categoriasDoDocumento(r.doc);
          log(`   tela de ${pagina.rotulo}: HTTP ${r.status}, ${selects} select(s), ${cats.length} categoria(s)`
            + `, contexto ${contextoDaPagina(r.html) || '—'}`);
          cats.slice(0, 12).forEach((c) => log(`      · "${c.nome}" cat=${c.valor} contexto=${c.contextid} grupo="${c.contexto}"`));
        }

        const cats = await categoriasDoBanco(banco.cmid, (m) => log(m));
        if (!cats.length) { log('   PAREI AQUI: nenhuma categoria.', 'erro'); break; }

        for (const cat of cats.slice(0, 3)) {
          for (const comFiltro of [true, false]) {
            const url = urlDaListagem(banco.cmid, cat, 0, comFiltro);
            const r = await baixaPagina(url);
            const d = r.doc;
            const caixas = d ? d.querySelectorAll('input[type="checkbox"][name^="q"]').length : 0;
            const previews = d ? d.querySelectorAll('a[href*="preview.php"]').length : 0;
            const linhas = d ? d.querySelectorAll('#questionscontainer tbody tr, .categoryquestionscontainer tbody tr, tbody tr').length : 0;
            const achadas = questoesDoDocumento(d);
            log(`   listagem "${cat.nome}" ${comFiltro ? 'COM' : 'SEM'} filtro: HTTP ${r.status}`
              + `, container ${d && d.querySelector('#questionscontainer') ? 'sim' : 'não'}`
              + `, ${linhas} linha(s), ${caixas} caixa(s) "q<id>", ${previews} link(s) de pré-visualização`
              + `, ${achadas.length} questão(ões) reconhecida(s)`, achadas.length ? 'ok' : 'aviso');

            // O Moodle embute no HTML o filtro que ele realmente entendeu.
            // É a resposta mais direta para saber se o filtro chegou certo.
            const marca = d && d.querySelector('[data-filtercondition]');
            if (marca) log(`      filtro entendido pelo Moodle: ${(marca.getAttribute('data-filtercondition') || '').slice(0, 400)}`);
            const aviso = d && (d.querySelector('.alert-danger, .errorbox, .notifyproblem'));
            if (aviso) log(`      AVISO NA PÁGINA: ${txt(aviso).slice(0, 300)}`, 'erro');
            if (!achadas.length && d) {
              const corpo = d.querySelector('#questionscontainer') || d.querySelector('#region-main') || d.body;
              log(`      texto da área das questões: "${txt(corpo).slice(0, 300)}"`);
            }
            if (achadas.length) {
              achadas.slice(0, 3).forEach((q) => log(`      · questão ${q.id} "${q.nome}"`));
              const alvo = achadas[0];
              const pv = await baixaPagina(urlDaPreVisualizacao(banco.cmid, alvo.id));
              const f = pv.doc && pv.doc.querySelector('.formulation');
              log(`      pré-visualização da questão ${alvo.id}: HTTP ${pv.status}`
                + `, .formulation ${f ? 'sim' : 'NÃO'}`
                + `, .qtext ${pv.doc && pv.doc.querySelector('.qtext') ? 'sim' : 'não'}`
                + `, ${pv.doc ? pv.doc.querySelectorAll('.answer .flex-fill').length : 0} alternativa(s)`,
                f ? 'ok' : 'erro');
            }
          }
        }
        break;
      }
      log('');
      log('Fim do diagnóstico. Copie este log e me mande.', 'ok');
    } catch (e) {
      log(`Parou: ${e.message}`, 'erro');
    } finally {
      rodando = false; interromper = false;
      travaBotoes(false); botaoParar.style.display = 'none'; botaoParar.textContent = 'Parar';
    }
  }

  // ----------------------------------------------------------------------
  // Painel
  // ----------------------------------------------------------------------
  // O painel se limita ao que cabe na tela: com a página ampliada, largura
  // fixa e altura livre deixavam o pé cortado, sem como rolar.
  const painel = document.createElement('div');
  painel.style.cssText = `position:fixed;top:10px;right:10px;z-index:99999;
    width:min(440px, calc(100vw - 20px));max-width:calc(100vw - 20px);
    max-height:calc(100vh - 20px);display:flex;flex-direction:column;
    font:12px/1.45 system-ui,sans-serif;background:#fff;border:1px solid #ccd;border-radius:8px;
    box-shadow:0 6px 20px rgba(0,0,0,.18);padding:10px;box-sizing:border-box;`;

  const titulo = document.createElement('div');
  titulo.style.cssText = 'font-weight:600;margin-bottom:8px;display:flex;align-items:center;gap:8px;flex:0 0 auto;';
  const tituloTexto = document.createElement('span');
  tituloTexto.textContent = 'Conferência das questões dos bancos';
  tituloTexto.style.cssText = 'flex:1;';
  const botaoEncolher = document.createElement('button');
  botaoEncolher.type = 'button';
  botaoEncolher.textContent = '–';
  botaoEncolher.title = 'Encolher / abrir o painel';
  botaoEncolher.style.cssText = `flex:0 0 auto;width:24px;height:24px;line-height:1;padding:0;cursor:pointer;
    border:1px solid #ccd;border-radius:5px;background:#f4f5f8;color:#334;font-size:14px;`;
  titulo.append(tituloTexto, botaoEncolher);

  const corpo = document.createElement('div');
  corpo.style.cssText = 'flex:1 1 auto;overflow-y:auto;overflow-x:hidden;min-height:0;';
  botaoEncolher.onclick = () => {
    const escondido = corpo.style.display === 'none';
    corpo.style.display = escondido ? '' : 'none';
    botaoEncolher.textContent = escondido ? '–' : '+';
  };

  const explicacao = document.createElement('div');
  explicacao.style.cssText = 'color:#556;margin-bottom:8px;';
  explicacao.textContent = 'Entra em cada sala desta categoria, abre os bancos de questões e procura: '
    + 'questão duplicada, alternativa repetida e imagem quebrada. Questão ou alternativa que é só imagem '
    + 'é comparada pelo conteúdo da imagem, então duas imagens diferentes não são acusadas de iguais. '
    + 'Nada é alterado — é só leitura. Se não achar questão, use "Diagnosticar (1 sala)" e me mande o log.';

  const botoes = document.createElement('div');
  botoes.style.cssText = 'display:flex;flex-wrap:wrap;gap:6px;';
  const criaBotao = (texto, cor, acao) => {
    const b = document.createElement('button');
    b.textContent = texto;
    b.style.cssText = `padding:8px 10px;background:${cor};color:#fff;border:0;border-radius:5px;cursor:pointer;`;
    b.onclick = acao;
    botoes.appendChild(b);
    return b;
  };

  const botao = criaBotao('Conferir as questões', '#28a745', verificar);
  const botaoDiagnosticar = criaBotao('Diagnosticar (1 sala)', '#6c757d', diagnosticar);
  const botaoCopiar = criaBotao('Copiar resumo', '#2d5bd0', copiarResumo);
  botaoCopiar.style.display = 'none';
  const botaoTxt = criaBotao('Baixar relatório (.txt)', '#17724d', baixarTxt);
  botaoTxt.style.display = 'none';
  const botaoParar = criaBotao('Parar', '#b3261e', () => { interromper = true; botaoParar.textContent = 'Parando…'; });
  botaoParar.style.display = 'none';

  const travaBotoes = (t) => { botao.disabled = t; botaoDiagnosticar.disabled = t; };

  const painelLog = document.createElement('div');
  painelLog.style.cssText = `margin-top:8px;max-height:45vh;overflow:auto;white-space:pre-wrap;
    overflow-wrap:anywhere;font-family:ui-monospace,monospace;font-size:11px;`;

  corpo.append(explicacao, botoes, painelLog);
  painel.append(titulo, corpo);
  document.body.appendChild(painel);

  function log(msg, tipo) {
    const cor = { ok: '#177245', erro: '#b3261e', aviso: '#8a6100' }[tipo] || '#333';
    const linha = document.createElement('div');
    linha.style.color = cor;
    linha.style.fontWeight = tipo ? '600' : 'normal';
    linha.textContent = msg;
    painelLog.appendChild(linha);
    painelLog.scrollTop = painelLog.scrollHeight;
    console.log('[QUESTOES]', msg);
  }
})();
