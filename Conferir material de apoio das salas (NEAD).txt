Diz quais salas têm material de apoio, quais estão vazias, quem é o professor de cada uma, detecta e corrige tópicos/atividades ocultos dos estudantes — e, dentro de Adaptação/Dependência, põe uma imagem numa seção nova no topo de todas as salas de um curso.

  'use strict';

  // ----------------------------------------------------------------------

  const CONFIG = {
    // Só para marcar no log se a sala segue o padrão esperado. Não afeta a
    // classificação "com conteúdo" / "sem conteúdo", que olha TODAS as seções.
    secoesPadrao: [

      /MATERIAL DE APOIO DA AV2\b/,

      /MATERIAL DE APOIO DA FINAL\b/,
    ],
    rotulosContamComoConteudo: false, // um mod_label (texto solto) conta como material?
    // Rótulo com link para fora do Moodle (Drive, YouTube…) conta: o material
    // está no link, não no Moodle.
    linksExternosContamComoConteudo: true,
    // Tipos de atividade que NÃO contam como material de estudo, mesmo que
    // existam na seção. Se só sobrar isso numa seção, ela é tratada como vazia.
    // Nomes técnicos do Moodle (o que vem depois de "modtype_" na classe do item).
    tiposIgnorados: ['forum'],
    // Professor da sala: lido na página de participantes do curso.
    mostrarProfessores: true,
    // Papéis que contam como professor. A comparação é feita sem acento, em
    // maiúsculas e sobre cada papel isoladamente, então /^PROFESSOR$/ casa com
    // "Professor" mas não com "Professor não editor" nem com "Tutor".
    // Para incluir outros papéis, acrescente aqui (ex.: /^PROFESSOR NAO EDITOR$/).
    papeisProfessor: [/^PROFESSOR(A)?$/],
    participantesPorPagina: 200,
    pausaMs: 600,
    usarIframe: true,                 // fallback quando a página do curso é montada por JS
    esperaIframeMs: 12000,

    // --- Conteúdo oculto (tópicos/atividades/recursos) ---------------------
    // Liga/desliga a nova checagem de visibilidade real do Moodle. Com false,
    // o script volta a se comportar exatamente como antes (nenhuma chamada
    // nova é feita).
    verificarOcultos: true,
    // true  = tenta tornar visível automaticamente o que estiver oculto.
    // false = só relata o que está oculto, sem alterar nada (modo "só relatório").
    // Também dá pra ligar/desligar isso pelo painel, na caixa de seleção.
    corrigirOcultos: true,
    // Pausa entre cada chamada de "tornar visível", pra não sobrecarregar o
    // servidor quando uma sala tiver muitos itens ocultos de uma vez.
    pausaAjaxMs: 350,
    // Mostra no console do navegador (F12) o estado bruto devolvido pelo
    // Moodle para a primeira sala processada — só pra conferir se os nomes de
    // campo usados aqui (visible, title, name…) batem com a sua instância,
    // caso algo pareça estranho no relatório.
    debugOcultos: true,
  };

  // ----------------------------------------------------------------------
  // Ajustes da funcionalidade "imagem no topo das salas"
  // ----------------------------------------------------------------------
  // Propositalmente separado do CONFIG acima: nada aqui interfere na
  // conferência de material de apoio.
  const CONFIG_IMAGEM = {
    // A funcionalidade só se habilita quando a categoria aberta é esta ou
    // está dentro dela. Comparado sem acento e sem diferença de maiúscula.
    categoriaRaiz: 'Adaptação/Dependência',
    // Nome da seção criada no topo da sala. Serve também de marca: sala que
    // já tiver uma seção com este nome é pulada, então rodar duas vezes não
    // duplica nada. Dá para trocar direto no painel.
    nomeDaSecao: 'AVISO',
    // Texto opcional colocado ABAIXO da imagem, em HTML
    // (ex.: '<p><strong>Prezado aluno,</strong></p>'). Dá para escrever
    // direto no painel; deixe vazio para pôr só a imagem.
    textoAbaixoDaImagem: '',
    // Quantos ids tentar quando for preciso descobrir o repositório
    // "Enviar um arquivo" perguntando ao Moodle, um a um.
    idsDeRepositorioParaSondar: 25,
    pausaEntreSalasMs: 900,
    pausaEntrePaginasMs: 250,
    pausaAjaxMs: 350,
  };

  const ORIGEM = location.origin;
  const dorme = (ms) => new Promise((r) => setTimeout(r, ms));
  const txt = (el) => (el && el.textContent || '').replace(/\u00a0/g, ' ').replace(/\s+/g, ' ').trim();
  const norm = (s) => s.normalize('NFKD').replace(/[\u0300-\u036f]/g, '').toUpperCase().replace(/\s+/g, ' ').trim();

  // ----------------------------------------------------------------------
  // Seções a ignorar — o usuário digita no campo de texto do painel.
  // Uma seção é ignorada quando o nome dela CONTÉM um dos termos digitados
  // (comparação sem acento e maiúsculas), então "Calendário de Provas" no
  // campo casa com a seção "CALENDÁRIO DE PROVAS" mesmo com pequenas variações.
  // ----------------------------------------------------------------------
  const CHAVE_LS = 'ugbAuditoriaSecoesIgnoradas';
  let secoesIgnoradas = []; // termos já normalizados

  function atualizaSecoesIgnoradas(textoCampo) {
    secoesIgnoradas = textoCampo
      .split(/[\n,;]+/)
      .map((s) => norm(s.trim()))
      .filter(Boolean);
  }

  function ehSecaoIgnorada(nome) {
    const n = norm(nome);
    return secoesIgnoradas.some((termo) => n.includes(termo));
  }

  // ----------------------------------------------------------------------
  // Rede
  // ----------------------------------------------------------------------
  async function baixaPagina(url) {
    try {
      const resp = await fetch(url, { credentials: 'same-origin' });
      const html = await resp.text();
      return { url, status: resp.status, ok: resp.ok, doc: new DOMParser().parseFromString(html, 'text/html') };
    } catch (e) {
      return { url, status: 0, ok: false, doc: null, erro: e.message };
    }
  }

  function carregaComJs(url, temOQuePreciso) {
    return new Promise((resolve) => {
      const frame = document.createElement('iframe');
      frame.style.cssText = 'position:fixed;left:-10000px;top:0;width:1280px;height:900px;border:0;';
      frame.src = url;
      document.body.appendChild(frame);
      const limite = Date.now() + CONFIG.esperaIframeMs;
      const timer = setInterval(() => {
        let doc = null;
        try { doc = frame.contentDocument; } catch (e) { /* ignora */ }
        if (doc && doc.readyState === 'complete' && temOQuePreciso(doc)) {
          clearInterval(timer);
          const copia = doc.cloneNode(true);
          setTimeout(() => frame.remove(), 100);
          resolve(copia);
          return;
        }
        if (Date.now() > limite) {
          clearInterval(timer);
          const copia = doc ? doc.cloneNode(true) : null;
          frame.remove();
          resolve(copia);
        }
      }, 400);
    });
  }

  // ----------------------------------------------------------------------
  // Visibilidade real (API do editor de cursos do Moodle)
  // ----------------------------------------------------------------------
  // Este bloco NÃO usa a mesma técnica do resto do script (ler o HTML da
  // página). Em vez disso, chama o mesmo endpoint AJAX que o editor de
  // cursos do Moodle usa para ler e alterar o estado real de seções e
  // atividades: o campo "visible" que vem daqui é o dado do banco
  // (course_sections.visible / course_modules.visible), não uma classe CSS.

  function pegaSesskey() {
    try {
      if (window.M && M.cfg && M.cfg.sesskey) return M.cfg.sesskey;
    } catch (e) { /* ignora */ }
    return null;
  }

  let _debugOcultosMostrado = false;

  // Chama o endpoint interno de AJAX do Moodle (lib/ajax/service.php) — o
  // mesmo que o editor de cursos usa. Autenticado pela sessão do navegador
  // (cookie enviado via credentials:'same-origin') e protegido por
  // "sesskey", o token anti-CSRF do Moodle.
  async function chamadaAjax(metodo, args) {
    const sesskey = pegaSesskey();
    if (!sesskey) throw new Error('sesskey não encontrado (M.cfg.sesskey ausente nesta página)');
    const url = `${ORIGEM}/lib/ajax/service.php?sesskey=${encodeURIComponent(sesskey)}&info=${encodeURIComponent(metodo)}`;
    const resp = await fetch(url, {
      method: 'POST',
      credentials: 'same-origin',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify([{ index: 0, methodname: metodo, args }]),
    });
    if (!resp.ok) throw new Error(`HTTP ${resp.status} ao chamar ${metodo}`);
    const payload = await resp.json();
    const item = Array.isArray(payload) ? payload[0] : null;
    if (!item) throw new Error(`resposta inesperada do servidor para ${metodo}`);
    if (item.error) {
      const msg = (item.exception && (item.exception.message || item.exception.errorcode)) || 'erro desconhecido';
      throw new Error(msg);
    }
    // O Moodle devolve o estado do curso como uma string JSON dentro de "data".
    return typeof item.data === 'string' ? JSON.parse(item.data) : item.data;
  }

  const ehFalso = (v) => v === false || v === 0 || v === '0';

  async function pegaEstadoCurso(courseid) {
    return chamadaAjax('core_courseformat_get_state', { courseid: Number(courseid) });
  }

  // acao: 'section_show' ou 'cm_show' (existem também 'section_hide' /
  // 'cm_hide', mas não são usadas aqui — o objetivo é só tornar visível).
  async function alteraVisibilidade(courseid, acao, id) {
    return chamadaAjax('core_courseformat_update_course', {
      action: acao, courseid: Number(courseid), ids: [Number(id)],
    });
  }

  function nomeEstadoSecao(s) {
    return s.title || s.name || `Seção nº ${s.number ?? s.id}`;
  }
  function nomeEstadoCm(c) {
    return c.name || `Item #${c.id}`;
  }

  // Lê o estado real do curso, identifica seções e atividades/recursos
  // marcados como ocultos (campo "visible", vindo do banco — não do HTML) e,
  // se CONFIG.corrigirOcultos estiver ligado, tenta tornar cada um visível.
  //
  // Seções ocultas são corrigidas primeiro: quando uma seção está oculta, os
  // itens dentro dela ficam indisponíveis para os estudantes mesmo que o
  // "visible" de cada item, individualmente, já esteja 1 — então só faz
  // sentido tentar mostrar um item cujo PRÓPRIO estado está oculto (os que só
  // estão "escondidos" por causa da seção voltam a aparecer sozinhos quando a
  // seção é mostrada, sem precisar mexer neles).
  async function verificaEAjustaOcultos(curso) {
    const resultado = { verificados: 0, ocultosEncontrados: [], alterados: [], falhas: [] };

    let estado;
    try {
      estado = await pegaEstadoCurso(curso.id);
    } catch (e) {
      resultado.falhas.push(`Não foi possível ler o estado real do curso: ${e.message}`);
      return resultado;
    }

    if (CONFIG.debugOcultos && !_debugOcultosMostrado) {
      _debugOcultosMostrado = true;
      console.log('[NEAD] Estado bruto devolvido pelo Moodle (1ª sala, para conferência):', estado);
    }

    const secoes = (estado && estado.section) || [];
    const cms = (estado && estado.cm) || [];
    resultado.verificados = secoes.length + cms.length;

    const secoesOcultas = secoes.filter((s) => ehFalso(s.visible));
    const cmsOcultos = cms.filter((c) => ehFalso(c.visible));

    secoesOcultas.forEach((s) => resultado.ocultosEncontrados.push({ tipo: 'Tópico/seção', nome: nomeEstadoSecao(s) }));
    cmsOcultos.forEach((c) => resultado.ocultosEncontrados.push({ tipo: 'Atividade/recurso', nome: nomeEstadoCm(c) }));

    if (!CONFIG.corrigirOcultos) return resultado;

    for (const s of secoesOcultas) {
      try {
        await alteraVisibilidade(curso.id, 'section_show', s.id);
        resultado.alterados.push({ tipo: 'Tópico/seção', nome: nomeEstadoSecao(s) });
      } catch (e) {
        resultado.falhas.push(`Seção "${nomeEstadoSecao(s)}": ${e.message}`);
      }
      await dorme(CONFIG.pausaAjaxMs);
    }

    for (const c of cmsOcultos) {
      try {
        await alteraVisibilidade(curso.id, 'cm_show', c.id);
        resultado.alterados.push({ tipo: 'Atividade/recurso', nome: nomeEstadoCm(c) });
      } catch (e) {
        resultado.falhas.push(`"${nomeEstadoCm(c)}": ${e.message}`);
      }
      await dorme(CONFIG.pausaAjaxMs);
    }

    return resultado;
  }

  // ----------------------------------------------------------------------
  // Cursos da categoria
  // ----------------------------------------------------------------------
  async function listarCursos() {
    const paginas = new Map([[location.href, document]]);
    document.querySelectorAll('.paging a[href], nav a[href*="page="]').forEach((a) => {
      if (!paginas.has(a.href)) paginas.set(a.href, null);
    });

    const cursos = new Map();
    for (const [url, cache] of paginas) {
      let doc = cache;
      if (!doc) { const r = await baixaPagina(url); if (!r.doc) continue; doc = r.doc; }
      const escopo = doc.querySelector('#region-main') || doc.body;
      escopo.querySelectorAll('a[href*="/course/view.php?id="]').forEach((a) => {
        const id = new URL(a.getAttribute('href'), ORIGEM).searchParams.get('id');
        const nome = txt(a);
        if (id && nome && !cursos.has(id)) cursos.set(id, nome);
      });
    }
    return [...cursos].map(([id, nome]) => ({ id, nome }));
  }

  // ----------------------------------------------------------------------
  const ROTULOS_MODTYPE = {
    resource: 'Arquivo', folder: 'Pasta', url: 'Link/URL', page: 'Página',

  function classificaPorTexto(texto) {
    const n = norm(texto || '');
    if (!n) return null;
    if (/\bPDF\b/.test(n)) return 'PDF';
    if (/POWERPOINT|APRESENTA/.test(n)) return 'Apresentação (PowerPoint)';
    if (/EXCEL|PLANILHA/.test(n)) return 'Planilha (Excel)';
    if (/\bWORD\b|DOCUMENTO DE TEXTO/.test(n)) return 'Documento (Word)';
    if (/IMAGEM|\bIMAGE\b/.test(n)) return 'Imagem';
    if (/V[IÍ]DEO/.test(n)) return 'Vídeo';
    if (/[AÁ]UDIO/.test(n)) return 'Áudio';
    if (/COMPACTADO|\bZIP\b|ARCHIVE/.test(n)) return 'Arquivo compactado';
    return null;
  }

  function capitaliza(s) { return s ? s.charAt(0).toUpperCase() + s.slice(1) : s; }

  function tipoDoItem(item, mtipo) {
    const nomeGenerico = ROTULOS_MODTYPE[mtipo] || (mtipo ? capitaliza(mtipo) : 'Item');
    if (mtipo !== 'resource') return nomeGenerico;

    const img =
      item.querySelector('.activityiconcontainer img, img.activityicon') ||
      item.querySelector('.activityinstance img, img[src*="/f/"]') ||
      item.querySelector('img');
    const alt = img ? (img.getAttribute('alt') || img.getAttribute('title') || '') : '';
    const src = img ? (img.getAttribute('src') || '') : '';
    const link = item.querySelector('a[href]');
    const href = link ? (link.getAttribute('href') || '') : '';

    const porAlt = classificaPorTexto(alt);
    if (porAlt) return porAlt;

    const mIcon = src.match(/\/f\/([a-z0-9_]+)(?:-\d+)?(?:[.?]|$)/i);
    if (mIcon && ROTULOS_FICON[mIcon[1].toLowerCase()]) return ROTULOS_FICON[mIcon[1].toLowerCase()];

    const mExt = href.match(/\.([a-zA-Z0-9]{2,5})(?:[?#]|$)/);
    if (mExt && ROTULOS_EXT[mExt[1].toLowerCase()]) return ROTULOS_EXT[mExt[1].toLowerCase()];

    return 'Arquivo';
  }

  function formataTipos(mapa) {
    return Object.entries(mapa)
      .sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0], 'pt'))
      .map(([tipo, n]) => `${tipo} (${n})`)
      .join(', ');
  }

  // Frases de acessibilidade/controle que às vezes ficam grudadas no título
  // da seção ("Selecionar seção", "Contrair", "Expandir" etc.), caso não
  // tenham vindo isoladas num elemento removível.
  const RE_CONTROLES_SECAO = /\b(Selecionar se[cç][aã]o|Editar (nome d)?a se[cç][aã]o|Contrair|Expandir|Collapse|Expand|Select section|Section actions|A[cç][oõ]es da se[cç][aã]o|Mostrar se[cç][aã]o|Ocultar se[cç][aã]o|Mover se[cç][aã]o)\b/gi;

  // Remove dos elementos de controle (menus, botões, checkboxes, texto só
  // para leitor de tela) antes de ler o texto, para não pegar "Selecionar
  // seção... Contrair Expandir..." junto com o nome de verdade.
  function limpaTituloSecao(el) {
    if (!el) return '';
    const clone = el.cloneNode(true);
    clone.querySelectorAll(
      '.controlmenu, .menu, .dropdown, .action-menu, button, [role="button"], input, ' +
      '.sr-only, .accesshide, .visually-hidden, .badge'
    ).forEach((n) => n.remove());
    return txt(clone).replace(RE_CONTROLES_SECAO, ' ').replace(/\s+/g, ' ').trim();
  }

  function extraiNomeSecao(sec) {
    // Do mais específico para o mais genérico — só cai no contêiner cheio
    // (.course-section-header) se nada mais específico existir.
    const candidatos = [
      sec.querySelector('h3.sectionname, h4.sectionname'),
      sec.querySelector('.sectionname h3, .sectionname h4'),
      sec.querySelector('[data-for="section_title"] h3, [data-for="section_title"] h4'),
      sec.querySelector('.sectionname'),
      sec.querySelector('[data-for="section_title"]'),
      sec.querySelector('.course-section-header'),
      sec.querySelector('h3, h4'),
    ];
    for (const el of candidatos) {
      const nome = limpaTituloSecao(el);
      if (nome) return nome;
    }
    return '';
  }


  // Um rótulo é texto solto e, por si, não é material de estudo. Mas quando ele
  // traz link para fora do Moodle — Drive, YouTube, um PDF em outro site —, é
  // ali que o material está: nesse caso o item conta como conteúdo.
  //
  // O mesmo vale para link que está na seção mas fora de qualquer atividade,
  // como no resumo/descrição da seção: para o aluno, é material do mesmo jeito.
  //
  // "Fora do Moodle" é decidido pelo endereço. Isso importa: a própria seção
  // tem links de menu, permalink e edição, todos do site — se qualquer link
  // contasse, toda seção passaria a ter conteúdo e a conferência perderia o
  // sentido.
  const HOSPEDEIRO = location.host;

  function ehEnderecoExterno(href) {
    if (!href || /^(#|mailto:|tel:|javascript:)/i.test(href)) return false;
    let alvo;
    try { alvo = new URL(href, ORIGEM); } catch (e) { return false; }
    if (!/^https?:$/.test(alvo.protocol)) return false;
    return !!alvo.host && alvo.host !== HOSPEDEIRO;
  }

  function linksExternosEm(no) {
    const achados = new Set();
    if (!no) return achados;
    no.querySelectorAll('a[href]').forEach((a) => {
      const href = a.getAttribute('href') || '';
      if (ehEnderecoExterno(href)) achados.add(new URL(href, ORIGEM).href);
    });
    // Endereço escrito no texto, sem virar link.
    const texto = txt(no) || '';
    (texto.match(/https?:\/\/[^\s<>"')]+/gi) || []).forEach((u) => {
      if (ehEnderecoExterno(u)) achados.add(u);
    });
    return achados;
  }

  const temLinkExterno = (item) => linksExternosEm(item).size > 0;

  // Links da seção que não estão dentro de nenhuma atividade (resumo da seção,
  // por exemplo). As atividades saem do clone porque já foram contadas uma a
  // uma, e os controles da seção saem para não entrarem como material.
  function linksExternosSoltos(secao) {
    const clone = secao.cloneNode(true);
    clone.querySelectorAll('li.activity, li.course-module, [data-for="cmitem"], .activity-item')
      .forEach((n) => n.remove());
    clone.querySelectorAll('.controlmenu, .menu, .dropdown, .action-menu, button, [role="button"], ' +
      '.sr-only, .accesshide, .visually-hidden, .section_action_menu, .sectionbadges')
      .forEach((n) => n.remove());
    return [...linksExternosEm(clone)];
  }


  // ----------------------------------------------------------------------
  // Leitura de TODAS as seções de um curso, seja qual for o nome delas
  // ----------------------------------------------------------------------
  function achaSecoes(doc) {
    if (!doc) return [];
    const nos = new Set();
    doc.querySelectorAll('li.section, li.course-section, [data-for="section"], .course-section').forEach((n) => nos.add(n));
    const secoes = [...nos].filter((n) => ![...nos].some((o) => o !== n && o.contains(n)));

    return secoes.map((sec) => {
      let nome = extraiNomeSecao(sec);
      if (!nome) {
        const idAttr = sec.getAttribute('data-number') ?? sec.getAttribute('aria-label') ?? '';
        nome = idAttr ? `Seção ${idAttr}` : 'Seção sem nome';
      }

      const itensNos = new Set();
      sec.querySelectorAll('li.activity, li.course-module, [data-for="cmitem"], .activity-item').forEach((n) => itensNos.add(n));
      const itens = [...itensNos].filter((n) => ![...itensNos].some((o) => o !== n && o.contains(n)));

      let conteudo = 0, rotulos = 0, ocultos = 0, ignoradosPorTipo = 0;
      const tiposConteudo = {};
      const itensVistos = [];
      for (const item of itens) {
        const classe = item.className || '';
        const mtipo = (classe.match(/modtype_(\w+)/) || [])[1] || '';
        const ehRotulo = /modtype_label|\blabel\b/.test(classe);
        const ehTipoIgnorado = CONFIG.tiposIgnorados.includes(mtipo.toLowerCase());
        itensVistos.push(mtipo || '(sem tipo)');
        const escondido = /\bhidden\b|dimmed/.test(classe) ||
          /OCULTO|N[ÃA]O DISPON[ÍI]VEL|HIDDEN FROM STUDENTS/.test(norm(txt(item.querySelector('.badge, .availabilityinfo')) || ''));
        if (escondido) ocultos++;
        const porLinkExterno = CONFIG.linksExternosContamComoConteudo && ehRotulo && temLinkExterno(item);
        if (ehTipoIgnorado) { ignoradosPorTipo++; continue; } // ex.: fórum não conta como material de estudo
        if (ehRotulo && !CONFIG.rotulosContamComoConteudo && !porLinkExterno) continue;
        if (!escondido) {
          conteudo++;
          const tipo = porLinkExterno ? 'Link externo' : tipoDoItem(item, mtipo);
          tiposConteudo[tipo] = (tiposConteudo[tipo] || 0) + 1;
        }
      }

      // Links que estão na seção mas fora das atividades — o resumo da seção,
      // tipicamente. Cada um conta como um material.
      const soltos = CONFIG.linksExternosContamComoConteudo ? linksExternosSoltos(sec) : [];
      if (soltos.length) {
        conteudo += soltos.length;
        tiposConteudo['Link externo'] = (tiposConteudo['Link externo'] || 0) + soltos.length;
      }

      const secaoOculta = /\bhidden\b|dimmed/.test(sec.className || '') ||
        /OCULTO PARA|HIDDEN FROM STUDENTS/.test(norm(txt(sec.querySelector('.badge, .section_availability')) || ''));

      return { nome, conteudo, rotulos, ocultos, ignoradosPorTipo, secaoOculta,
               totalItens: itens.length, tiposConteudo, itensVistos, linksSoltos: soltos.length };
    });
  }

  // ----------------------------------------------------------------------
  // Professores da sala (página de participantes do curso)
  // ----------------------------------------------------------------------
  function temTabelaParticipantes(doc) {
    return !!doc && !!doc.querySelector('#participants, #participantsform, table#participants, [data-table-uniqueid*="participants"]');
  }

  function linhasParticipantes(doc) {
    if (!doc) return [];
    const tabela =
      doc.querySelector('table#participants') ||
      doc.querySelector('#participantsform table') ||
      doc.querySelector('#region-main table.generaltable') ||
      doc.querySelector('table.generaltable');
    if (!tabela) return [];
    return [...tabela.querySelectorAll('tbody tr')].filter((tr) => linksDeParticipante(tr).length);
  }

  // O avatar do participante é um link para o perfil igual ao do nome, e
  // quando o usuário não tem foto o Moodle desenha as iniciais como TEXTO
  // ("FD"). Sem tirar isso, o "nome" do professor viraria as iniciais — e dois
  // professores sem foto virariam um só.
  const SELETOR_AVATAR = '.userpicture, .userinitials, .avatar, .initials, img, svg';

  function linksDeParticipante(linha) {
    return [...linha.querySelectorAll('a[href*="/user/view.php"], a[href*="/user/profile.php"]')];
  }

  function nomeDoParticipante(linha) {
    for (const link of linksDeParticipante(linha)) {
      const clone = link.cloneNode(true);
      clone.querySelectorAll(SELETOR_AVATAR).forEach((n) => n.remove());
      const nome = txt(clone);
      if (nome) return nome;
    }
    return '';
  }

  function ehPapelProfessor(papel) {
    const n = norm(papel);
    return !!n && CONFIG.papeisProfessor.some((re) => re.test(n));
  }

  // Ícones e textos de leitor de tela que ficam grudados no nome do papel
  // ("Papéis", "Editar papéis", o lápis de edição) e estragariam a comparação
  // exata. Atenção: NÃO dá para remover os links aqui — no "inplace editable"
  // do Moodle o nome do papel fica dentro do próprio link de edição.
  const SELETOR_CONTROLES_CELULA =
    '.quickediticon, .editinstructions, img, i.icon, .sr-only, .sr-only-focusable, ' +
    '.accesshide, .visually-hidden, button, input, select, .action-menu, .dropdown, .menu';

  // Devolve os papéis do participante, um por item da lista.
  // A coluna "Papéis" aparece em três formatos, dependendo da versão/tema:
  //   1. inplace editable — <span class="displayvalue">Professor</span> dentro
  //      do link do lápis (é o formato do Moodle 4.x/5 em uso aqui);
  //   2. um badge/li por papel;
  //   3. texto solto, com os papéis separados por vírgula.
  function papeisDaCelula(celula) {
    const clone = celula.cloneNode(true);
    clone.querySelectorAll(SELETOR_CONTROLES_CELULA).forEach((n) => n.remove());

    const editaveis = [...clone.querySelectorAll('.displayvalue')];
    const marcados = editaveis.length ? editaveis : [...clone.querySelectorAll('.badge, li, .role')];
    const textos = marcados.length ? marcados.map((n) => txt(n)) : [txt(clone)];

    return textos
      .flatMap((t) => t.split(/[,;\n\u00b7|]+/))
      .map((t) => t.trim())
      .filter(Boolean);
  }

  // Além dos nomes, devolvo os papéis que apareceram na tabela: quando nenhum
  // professor é encontrado, é isso que mostra se o problema foi a leitura da
  // página ou se a sala realmente não tem ninguém com o papel esperado.
  function extraiProfessores(linhas) {
    const nomes = [];
    const papeisVistos = new Set();
    for (const linha of linhas) {
      const nome = nomeDoParticipante(linha);
      if (!nome) continue;
      const celulasDoNome = linksDeParticipante(linha).map((a) => a.closest('td, th'));
      const celulas = [...linha.querySelectorAll('td, th')].filter((c) => !celulasDoNome.includes(c));

      let ehProfessor = false;
      for (const celula of celulas) {
        for (const papel of papeisDaCelula(celula)) {
          if (!ehPapelProfessor(papel)) continue;
          ehProfessor = true;
          papeisVistos.add(papel);
        }
      }
      if (!ehProfessor) {
        const celulaPapeis = celulas.find((c) => c.querySelector('[data-itemtype="user_roles"], [data-region="roles"]'));
        if (celulaPapeis) papeisDaCelula(celulaPapeis).forEach((p) => papeisVistos.add(p));
        continue;
      }
      if (!nomes.includes(nome)) nomes.push(nome);
    }
    return {
      nomes: nomes.sort((a, b) => a.localeCompare(b, 'pt')),
      papeisVistos: [...papeisVistos],
    };
  }

  // Devolve { nomes, lido }: "lido" separa "a sala não tem professor" de
  // "não consegui abrir a lista de participantes" (sem permissão, erro etc.).
  async function listaProfessores(courseid) {
    const url = `${ORIGEM}/user/index.php?id=${courseid}&perpage=${CONFIG.participantesPorPagina}`;
    const r = await baixaPagina(url);
    let doc = r.doc;
    let linhas = linhasParticipantes(doc);

    // Sem nenhuma linha, tento de novo dentro de um iframe: em parte das
    // versões a tabela de participantes só é montada por JavaScript.
    if (!linhas.length && CONFIG.usarIframe) {
      const comJs = await carregaComJs(url, (d) => linhasParticipantes(d).length > 0);
      if (comJs && linhasParticipantes(comJs).length) { doc = comJs; linhas = linhasParticipantes(comJs); }
    }

    if (!linhas.length) return { nomes: [], papeisVistos: [], lido: temTabelaParticipantes(doc) };
    return { ...extraiProfessores(linhas), lido: true };
  }

  function textoProfessores(r) {
    const p = (r && r.professores) || { nomes: [], papeisVistos: [], lido: false };
    if (!p.lido) return 'não foi possível ler os participantes';
    if (p.nomes.length) return p.nomes.join(', ');
    const vistos = (p.papeisVistos || []).join(', ');
    return vistos ? `sem professor atribuído (papéis na sala: ${vistos})` : 'sem professor atribuído';
  }

  // Resume o resultado da checagem de conteúdo oculto de uma sala, pra uso
  // direto no log linha-a-linha.
  function linhaOcultos(o) {
    if (!o) return '';
    if (o.falhas.length && !o.ocultosEncontrados.length) {
      return `Conteúdo oculto: não foi possível verificar (${o.falhas[0]})`;
    }
    if (!o.ocultosEncontrados.length) return 'Conteúdo oculto: nenhum encontrado.';
    const detalhe = o.ocultosEncontrados.map((x) => `${x.tipo}: ${x.nome}`).join(' · ');
    if (!CONFIG.corrigirOcultos) {
      return `Conteúdo oculto encontrado (${o.ocultosEncontrados.length}, não corrigido — modo relatório): ${detalhe}`;
    }
    const falhaTxt = o.falhas.length ? ` — falhas: ${o.falhas.join(' | ')}` : '';
    return `Conteúdo oculto encontrado (${o.ocultosEncontrados.length}), tornado visível (${o.alterados.length}): ${detalhe}${falhaTxt}`;
  }

  async function leCurso(curso) {
    const url = `${ORIGEM}/course/view.php?id=${curso.id}`;
    const r = await baixaPagina(url);
    let brutas = achaSecoes(r.doc);

    if (!brutas.length && CONFIG.usarIframe) {
      const doc = await carregaComJs(url, (d) => achaSecoes(d).length > 0);
      brutas = achaSecoes(doc);
    }

    const professores = CONFIG.mostrarProfessores
      ? await listaProfessores(curso.id)
      : { nomes: [], papeisVistos: [], lido: false };

    const ocultos = CONFIG.verificarOcultos
      ? await verificaEAjustaOcultos(curso)
      : null;

    const ignoradas = brutas.filter((s) => ehSecaoIgnorada(s.nome));
    const secoes = brutas.filter((s) => !ehSecaoIgnorada(s.nome));
    return { url, status: r.status, secoes, totalBruto: brutas.length, ignoradas: ignoradas.map((s) => s.nome), professores, ocultos };
  }

  // ----------------------------------------------------------------------
  // Avaliação: olha TODAS as seções que sobraram após o filtro, sem
  // filtrar por nome esperado — só o que o usuário marcou para ignorar sai.
  // ----------------------------------------------------------------------
  function avalia(curso, leitura) {
    const secoes = leitura.secoes.map((s) => {
      let estado;
      if (s.secaoOculta) estado = 'oculta';
      else if (s.conteudo === 0) estado = 'vazia';
      else estado = 'com conteúdo';
      return { nome: s.nome, estado, itens: s.conteudo, ocultos: s.ocultos,
               ignoradosPorTipo: s.ignoradosPorTipo || 0, tiposConteudo: s.tiposConteudo || {},
               itensVistos: s.itensVistos || [], linksSoltos: s.linksSoltos || 0 };
    });

    const comConteudo = secoes.filter((s) => s.estado === 'com conteúdo');
    const seguePadrao = CONFIG.secoesPadrao.every((re) =>
      leitura.secoes.some((s) => re.test(norm(s.nome)) && !s.secaoOculta && s.conteudo > 0));

    // "ilegível" só quando a página não deu pra ler NADA; se sobrou 0 seções
    // porque tudo foi ignorado, a sala é avaliada normalmente (fica "sem conteúdo").
    let estado;
    if (!leitura.totalBruto) estado = 'ilegível';
    else if (comConteudo.length) estado = 'com conteúdo';
    else estado = 'sem conteúdo';

    return {
      curso, url: leitura.url, secoes, estado, seguePadrao,
      ignoradas: leitura.ignoradas || [],
      professores: leitura.professores || { nomes: [], lido: false },
      ocultos: leitura.ocultos || null,
    };
  }

  // ----------------------------------------------------------------------
  // Execução
  // ----------------------------------------------------------------------
  let rodando = false, interromper = false;
  let relatorio = [];

  async function verificar() {
    if (rodando) return;
    rodando = true; interromper = false; relatorio = [];
    travaBotoes(true); botaoParar.style.display = ''; botaoCopiar.style.display = 'none';
    painelLog.innerHTML = '';

    try {
      const cursos = await listarCursos();
      log(`${cursos.length} salas nesta categoria.`);

      for (const curso of cursos) {
        if (interromper) { log('Interrompido por você.', 'aviso'); break; }
        const curto = curso.nome.replace(/\[[^\]]*\]\s*/g, '').replace(/\s*[•⬩].*$/, '').trim();

        try {
          const leitura = await leCurso(curso);
          const r = avalia(curso, leitura);
          relatorio.push(r);

          const tipo = r.estado === 'com conteúdo' ? 'ok' : (r.estado === 'ilegível' ? 'erro' : 'aviso');
          log(curto, tipo);

          if (CONFIG.mostrarProfessores) log(`   Professor(es): ${textoProfessores(r)}`);

          if (CONFIG.verificarOcultos && r.ocultos) {
            const tipoOcultos = r.ocultos.falhas.length
              ? 'erro'
              : (r.ocultos.ocultosEncontrados.length ? 'aviso' : undefined);
            log(`   ${linhaOcultos(r.ocultos)}`, tipoOcultos);
          }

          if (r.ignoradas.length) log(`   (ignorando: ${r.ignoradas.join(' | ')})`);

          if (r.secoes.length) {
            const linhas = r.secoes.map((s) => {
              const nota = s.ignoradosPorTipo ? ` (${s.ignoradosPorTipo} item(ns) de tipo ignorado, ex.: fórum)` : '';
              if (s.estado === 'oculta') return `${s.nome}: oculta (${s.itens} itens)${nota}`;
              // Numa seção vazia, digo o que foi visto: é isso que mostra se a
              // conferência errou ou se a seção está vazia mesmo.
              if (s.estado === 'vazia') {
                const vistos = s.itensVistos.length ? s.itensVistos.join(', ') : 'nenhum item';
                return `${s.nome}: vazia (vi: ${vistos}; ${s.linksSoltos} link(s) externo(s) fora das atividades)${nota}`;
              }
              const composicao = formataTipos(s.tiposConteudo);
              return `${s.nome}: ${s.itens} item(ns) — ${composicao}${s.ocultos ? ` (+${s.ocultos} ocultos)` : ''}${nota}`;
            });
            log(`   ${linhas.join(' · ')}`, tipo);
          } else if (r.ignoradas.length) {
            log('   (nenhuma seção restante depois de ignorar as marcadas acima)', tipo);
          }
          log(`   → ${r.estado}${r.seguePadrao ? ' · estrutura padrão' : ''}`, tipo);
        } catch (e) {
          log(`${curto}: erro ao ler (${e.message})`, 'erro');
          relatorio.push({
            curso, url: `${ORIGEM}/course/view.php?id=${curso.id}`, estado: 'erro',
            secoes: [], seguePadrao: false, professores: { nomes: [], papeisVistos: [], lido: false },
            ocultos: null,
          });
        }
        await dorme(CONFIG.pausaMs);
      }

      resumoFinal();
      if (relatorio.length) botaoCopiar.style.display = '';
    } catch (e) {
      log(`Parou: ${e.message}`, 'erro');
    } finally {
      rodando = false; interromper = false;
      travaBotoes(false); botaoParar.style.display = 'none'; botaoParar.textContent = 'Parar';
    }
  }

  function nomeCurto(curso) {
    return curso.nome.replace(/\[[^\]]*\]\s*/g, '').replace(/\s*[•⬩].*$/, '').trim();
  }

  // Uma linha por sala no resumo, já com o(s) professor(es) da matéria.
  function linhaSala(indentacao, r) {
    const nome = nomeCurto(r.curso);
    if (!CONFIG.mostrarProfessores) return `${indentacao}- ${nome}`;
    return `${indentacao}- ${nome} — Professor(es): ${textoProfessores(r)}`;
  }

  function linhasDetalheSala(r) {
    return r.secoes
      .filter((s) => s.estado === 'com conteúdo')
      .map((s) => `      · ${s.nome}: ${formataTipos(s.tiposConteudo)}`);
  }

  // Bloco final do resumo com o total de conteúdo oculto encontrado/corrigido
  // em todas as salas, e a lista de salas que precisam de atenção manual
  // (erro de permissão, item que o Moodle recusou alterar etc.).
  function linhasResumoOcultos() {
    let encontrados = 0, corrigidos = 0;
    const comFalha = [];
    relatorio.forEach((r) => {
      if (!r.ocultos) return;
      encontrados += r.ocultos.ocultosEncontrados.length;
      corrigidos += r.ocultos.alterados.length;
      if (r.ocultos.falhas.length) comFalha.push({ nome: nomeCurto(r.curso), falhas: r.ocultos.falhas });
    });
    const linhas = [''];
    linhas.push(CONFIG.corrigirOcultos
      ? `CONTEÚDO OCULTO — ${encontrados} encontrado(s), ${corrigidos} tornado(s) visível`
      : `CONTEÚDO OCULTO — ${encontrados} encontrado(s) (modo relatório: nada foi alterado)`);
    if (comFalha.length) {
      linhas.push(`  precisam de atenção manual (${comFalha.length} sala(s)):`);
      comFalha.forEach((c) => linhas.push(`    - ${c.nome}: ${c.falhas.join(' | ')}`));
    }
    return linhas;
  }

  function textoResumo() {
    const comConteudo = relatorio.filter((r) => r.estado === 'com conteúdo');
    const semConteudo = relatorio.filter((r) => r.estado === 'sem conteúdo');
    const ilegiveis = relatorio.filter((r) => r.estado === 'ilegível' || r.estado === 'erro');

    const padrao = comConteudo.filter((r) => r.seguePadrao);
    const foraPadrao = comConteudo.filter((r) => !r.seguePadrao);

    const linhas = [];
    linhas.push(`SALAS COM CONTEÚDO (${comConteudo.length})`);
    if (padrao.length) {
      linhas.push(`  estrutura padrão (${padrao.length}):`);
      padrao.forEach((r) => { linhas.push(linhaSala('    ', r)); linhas.push(...linhasDetalheSala(r)); });
    }
    if (foraPadrao.length) {
      linhas.push(`  fora do padrão, mas com material (${foraPadrao.length}):`);
      foraPadrao.forEach((r) => { linhas.push(linhaSala('    ', r)); linhas.push(...linhasDetalheSala(r)); });
    }
    linhas.push('');
    linhas.push(`SALAS SEM CONTEÚDO (${semConteudo.length})`);
    semConteudo.forEach((r) => linhas.push(linhaSala('  ', r)));
    if (ilegiveis.length) {
      linhas.push('');
      linhas.push(`NÃO FOI POSSÍVEL VERIFICAR (${ilegiveis.length})`);
      ilegiveis.forEach((r) => linhas.push(linhaSala('  ', r)));
    }

    if (CONFIG.verificarOcultos) linhas.push(...linhasResumoOcultos());

    return linhas.join('\n');
  }

  function resumoFinal() {
    log('');
    log('════════ RESUMO ════════');
    for (const linha of textoResumo().split('\n')) {
      const tipo = linha.startsWith('SALAS COM') ? 'ok' : (linha.startsWith('SALAS SEM') || linha.startsWith('NÃO FOI') ? 'erro' : undefined);
      log(linha, tipo);
    }
  }

  function copiarResumo() {
    navigator.clipboard.writeText(textoResumo())
      .then(() => log('Resumo copiado para a área de transferência.', 'ok'))
      .catch(() => log('Não consegui copiar automaticamente — selecione o texto no painel.', 'aviso'));
  }

  // ----------------------------------------------------------------------
  // Imagem no topo das salas (somente Adaptação/Dependência)
  // ----------------------------------------------------------------------
  // Este bloco é independente de tudo que está acima: a conferência de
  // material de apoio continua exatamente como era. O que ele faz, em ordem:
  //   1. descobre todas as salas que existem dentro da(s) categoria(s)
  //      marcada(s) no painel (entrando também nas subcategorias);
  //   2. cria uma seção nova logo abaixo de "Geral", ou seja, no topo da
  //      página da sala;
  //   3. envia a imagem escolhida e a deixa aparecendo dentro dessa seção.
  // Sala que já tiver uma seção com o nome configurado é pulada.

  const SEPARADOR_CATEGORIA = ' / ';

  // O caminho vem do Moodle como "Graduação / 2026.2 / Campus Volta Redonda /
  // Adaptação/Dependência / Administração". Separo pelo " / " (com espaços),
  // justamente para não quebrar o nome "Adaptação/Dependência" ao meio.
  function segmentosDoCaminho(caminho) {
    return String(caminho || '').split(SEPARADOR_CATEGORIA).map((s) => s.trim()).filter(Boolean);
  }

  function dentroDaRaizDependencia(caminho) {
    const alvo = norm(CONFIG_IMAGEM.categoriaRaiz);
    return segmentosDoCaminho(caminho).some((s) => norm(s) === alvo);
  }

  function idDaCategoriaAtual() {
    try {
      const daUrl = new URL(location.href).searchParams.get('categoryid');
      if (daUrl) return String(daUrl);
    } catch (e) { /* ignora */ }
    const m = (document.body.className || '').match(/\bcategory-(\d+)\b/);
    return m ? m[1] : null;
  }

  // A página de categoria do Moodle traz um <select name="jump"> com TODAS as
  // categorias do site e o caminho completo de cada uma. É a forma mais barata
  // de saber quem é filha de quem: uma requisição só, sem paginação.
  function mapaDeCategorias(doc) {
    const d = doc || document;
    const mapa = new Map();
    d.querySelectorAll('select[name="jump"] option').forEach((op) => {
      const valor = op.getAttribute('value') || '';
      let id = null;
      const m = valor.match(/categoryid=(\d+)/);
      if (m) id = m[1];
      else if (/^\d+$/.test(valor.trim())) id = valor.trim();
      if (!id) return;
      const caminho = txt(op);
      if (caminho && !mapa.has(id)) mapa.set(id, caminho);
    });
    return mapa;
  }

  // Sem o <select>, monto o caminho pela trilha de navegação da própria página.
  function caminhoPelaTrilha() {
    const partes = [];
    document.querySelectorAll('.breadcrumb a[href*="categoryid="], nav[aria-label] a[href*="categoryid="]')
      .forEach((a) => { const t = txt(a); if (t && !partes.includes(t)) partes.push(t); });
    const h1 = txt(document.querySelector('#region-main h1, h1'));
    if (h1 && !partes.includes(h1)) partes.push(h1);
    return partes.join(SEPARADOR_CATEGORIA);
  }

  function filhasDiretas(caminhoPai, mapa) {
    const prefixo = caminhoPai + SEPARADOR_CATEGORIA;
    const filhas = [];
    for (const [id, caminho] of mapa) {
      if (!caminho.startsWith(prefixo)) continue;
      const resto = caminho.slice(prefixo.length);
      if (resto.includes(SEPARADOR_CATEGORIA)) continue; // neta, não filha
      filhas.push({ id, nome: resto, caminho });
    }
    filhas.sort((a, b) => a.nome.localeCompare(b.nome, 'pt-BR'));
    return filhas;
  }

  // Descobre onde o script está sendo usado e o que pode ser oferecido no
  // painel. Devolve sempre um objeto, mesmo quando a página não serve.
  function contextoDaCategoria() {
    const id = idDaCategoriaAtual();
    if (!id) return { ok: false, motivo: 'Esta página não é a de uma categoria de cursos.' };

    const mapa = mapaDeCategorias(document);
    const caminho = mapa.get(id) || caminhoPelaTrilha();
    if (!caminho) return { ok: false, motivo: 'Não consegui descobrir o nome desta categoria.' };

    if (!dentroDaRaizDependencia(caminho)) {
      return {
        ok: false,
        motivo: `Esta função só vale para "${CONFIG_IMAGEM.categoriaRaiz}". Você está em: ${caminho}`,
      };
    }

    let filhas = mapa.size ? filhasDiretas(caminho, mapa) : [];
    if (!filhas.length) {
      // Categoria folha (já é o curso em si) ou site sem o <select>: ofereço a
      // própria categoria aberta.
      const nome = segmentosDoCaminho(caminho).pop() || caminho;
      filhas = [{ id, nome: `${nome} (esta categoria)`, caminho }];
    }
    return { ok: true, id, caminho, mapa, filhas };
  }

  // ----------------------------------------------------------------------
  // Salas de uma categoria (entrando nas subcategorias)
  // ----------------------------------------------------------------------
  async function baixaPaginaComHtml(url, opcoes) {
    try {
      const resp = await fetch(url, Object.assign({ credentials: 'same-origin' }, opcoes || {}));
      const html = await resp.text();
      return {
        url: resp.url || url, status: resp.status, ok: resp.ok, html,
        doc: new DOMParser().parseFromString(html, 'text/html'),
      };
    } catch (e) {
      return { url, status: 0, ok: false, html: '', doc: null, erro: e.message };
    }
  }

  // Só aceito como subcategoria o bloco que o Moodle desenha na listagem
  // (div.category[data-categoryid]). Link solto com "categoryid=" não serve:
  // poderia ser o link de volta para a categoria pai e a varredura subiria a
  // árvore inteira, pegando salas que não foram pedidas.
  function subcategoriasNaPagina(doc) {
    const escopo = doc.querySelector('#region-main') || doc.body;
    const ids = [];
    escopo.querySelectorAll('.category[data-categoryid]').forEach((el) => {
      const id = el.getAttribute('data-categoryid');
      if (id && !ids.includes(id)) ids.push(id);
    });
    return ids;
  }

  async function salasDaCategoria(categoryid, contexto, anota) {
    const salas = new Map();
    const fila = [String(categoryid)];
    const categoriasVistas = new Set(fila);
    const caminhoRaiz = contexto && contexto.mapa ? contexto.mapa.get(String(categoryid)) : null;

    while (fila.length) {
      if (interromper) break;
      const cat = fila.shift();
      const paginas = [`${ORIGEM}/course/index.php?categoryid=${encodeURIComponent(cat)}`];
      const paginasVistas = new Set(paginas);

      while (paginas.length) {
        if (interromper) break;
        const url = paginas.shift();
        const r = await baixaPagina(url);
        if (!r.doc) { if (anota) anota(`   não consegui abrir ${url}`); continue; }
        const escopo = r.doc.querySelector('#region-main') || r.doc.body;

        escopo.querySelectorAll('a[href*="/course/view.php?id="]').forEach((a) => {
          const id = new URL(a.getAttribute('href'), ORIGEM).searchParams.get('id');
          const nome = txt(a);
          if (id && nome && !salas.has(id)) salas.set(id, nome);
        });

        subcategoriasNaPagina(r.doc).forEach((id) => {
          if (categoriasVistas.has(id)) return;
          // Trava extra: quando conheço a árvore, só entro em quem realmente
          // está abaixo da categoria escolhida.
          if (caminhoRaiz) {
            const caminhoFilha = contexto.mapa.get(id);
            if (caminhoFilha && !caminhoFilha.startsWith(caminhoRaiz + SEPARADOR_CATEGORIA)) return;
          }
          categoriasVistas.add(id);
          fila.push(id);
        });

        // A listagem vem paginada (20 por página): sigo os links de paginação.
        escopo.querySelectorAll('.paging a[href], nav a[href*="page="]').forEach((a) => {
          const destino = new URL(a.getAttribute('href'), ORIGEM).href;
          if (!destino.includes('categoryid=')) return;
          if (paginasVistas.has(destino)) return;
          paginasVistas.add(destino);
          paginas.push(destino);
        });

        await dorme(CONFIG_IMAGEM.pausaEntrePaginasMs);
      }
    }

    return [...salas].map(([id, nome]) => ({ id, nome }));
  }

  // ----------------------------------------------------------------------
  // Criar a seção e pôr a imagem dentro

  function nomeCruDaSecao(s) {
    return (s && (s.title || s.name)) ? String(s.title || s.name) : '';
  }

  function secaoJaExistente(estado, nomeProcurado) {
    const alvo = norm(nomeProcurado);
    if (!alvo) return null;
    return ((estado && estado.section) || []).find((s) => norm(nomeCruDaSecao(s)) === alvo) || null;
  }

  function numeroDaSecao(s) {
    const n = Number(s.number);
    return Number.isFinite(n) ? n : NaN;
  }

  // Cria a seção logo depois da seção 0 ("Geral"), que é o topo da página da
  // sala. O Moodle faz isso sozinho quando recebe targetsectionid: a nova
  // seção entra na posição seguinte à indicada.
  async function criaSecaoNoTopo(courseid, estadoAntes) {
    const secoes = (estadoAntes && estadoAntes.section) || [];
    const geral = secoes.find((s) => numeroDaSecao(s) === 0);
    if (!geral) throw new Error('não achei a seção "Geral" (seção 0) desta sala');

    const idsAntes = new Set(secoes.map((s) => String(s.id)));

    await chamadaAjax('core_courseformat_update_course', {
      action: 'section_add', courseid: Number(courseid), ids: [], targetsectionid: Number(geral.id),
    });

    const estadoDepois = await pegaEstadoCurso(courseid);
    const novas = ((estadoDepois && estadoDepois.section) || []).filter((s) => !idsAntes.has(String(s.id)));
    if (!novas.length) throw new Error('a seção foi pedida, mas o Moodle não devolveu nenhuma seção nova');
    return { secao: novas[0], estado: estadoDepois };
  }

  // O id da instância do repositório "Enviar um arquivo" é o mesmo no site
  // inteiro, então descubro uma vez só e guardo. Procuro em três lugares, do
  // mais barato para o mais caro, porque o formato da página muda conforme o
  // editor de texto instalado (Atto, TinyMCE…) e eu não quero depender disso.
  let _repositorioDeEnvio = null;

  function procuraRepositorioNoHtml(html) {
    if (!html) return null;
    // Em algumas telas o JSON vem escapado dentro de um atributo HTML.
    const textos = [html, html.replace(/&quot;/g, '"').replace(/&amp;/g, '&')];
    for (const texto of textos) {
      const direto = texto.match(/"id"\s*:\s*(\d+)\s*,\s*"name"\s*:\s*"(?:[^"\\]|\\.)*"\s*,\s*"type"\s*:\s*"upload"/);
      if (direto) return direto[1];
      const blocos = texto.match(/\{[^{}]*"type"\s*:\s*"upload"[^{}]*\}/g) || [];
      for (const bloco of blocos) {
        const m = bloco.match(/"id"\s*:\s*(\d+)/);
        if (m) return m[1];
      }
    }
    return null;
  }

  // Última tentativa: pergunta ao próprio Moodle. O repositório "Enviar um
  // arquivo" é o único que responde com um formulário de envio (upload.id
  // "repo-form"), então dá para reconhecê-lo testando os ids um a um.
  async function sondaRepositorioDeEnvio(ctxId) {
    const sesskey = pegaSesskey();
    if (!sesskey) return null;
    for (let id = 1; id <= CONFIG_IMAGEM.idsDeRepositorioParaSondar; id++) {
      const dados = new FormData();
      dados.append('sesskey', sesskey);
      dados.append('repo_id', String(id));
      if (ctxId) dados.append('ctx_id', String(ctxId));
      dados.append('env', 'editor');
      let json = null;
      try {
        const resp = await fetch(`${ORIGEM}/repository/repository_ajax.php?action=list`, {
          method: 'POST', credentials: 'same-origin', body: dados,
        });
        json = JSON.parse(await resp.text());
      } catch (e) { continue; }
      if (json && json.upload && json.upload.id === 'repo-form') return String(id);
    }
    return null;
  }

  async function idDoRepositorioDeEnvio(htmlDaPagina, ctxId, anota) {
    if (_repositorioDeEnvio) return _repositorioDeEnvio;

    let id = procuraRepositorioNoHtml(htmlDaPagina);
    if (id) { _repositorioDeEnvio = id; if (anota) anota(`   (repositório de envio nº ${id}, achado na própria tela)`); return id; }

    // "Arquivos privados" tem um gerenciador de arquivos e lista os
    // repositórios no HTML em qualquer versão do Moodle.
    const privados = await baixaPaginaComHtml(`${ORIGEM}/user/files.php`);
    id = procuraRepositorioNoHtml(privados.html);
    if (id) { _repositorioDeEnvio = id; if (anota) anota(`   (repositório de envio nº ${id}, achado em Arquivos privados)`); return id; }

    id = await sondaRepositorioDeEnvio(ctxId);
    if (id) { _repositorioDeEnvio = id; if (anota) anota(`   (repositório de envio nº ${id}, achado perguntando ao Moodle)`); return id; }

    throw new Error('não achei o repositório "Enviar um arquivo". Dica: use o campo de endereço (URL) da imagem no painel');
  }

  function achaContextId(html) {
    const m = html.match(/"contextid"\s*:\s*(\d+)/);
    return m ? m[1] : null;
  }

  // Manda o arquivo para a área de rascunho do usuário (o mesmo lugar onde o
  // Moodle guarda um arquivo recém-enviado antes de salvar o formulário).
  async function enviaArquivoParaRascunho(arquivo, itemid, repoId, ctxId) {
    const sesskey = pegaSesskey();
    if (!sesskey) throw new Error('sesskey não encontrado nesta página');

    const dados = new FormData();
    dados.append('sesskey', sesskey);
    dados.append('repo_id', String(repoId));
    dados.append('itemid', String(itemid));
    dados.append('ctx_id', String(ctxId));
    dados.append('env', 'editor');
    dados.append('savepath', '/');
    dados.append('title', arquivo.name);
    dados.append('overwrite', '1');
    dados.append('author', '');
    dados.append('repo_upload_file', arquivo, arquivo.name);

    const resp = await fetch(`${ORIGEM}/repository/repository_ajax.php?action=upload`, {
      method: 'POST', credentials: 'same-origin', body: dados,
    });
    const bruto = await resp.text();
    let json;
    try {
      json = JSON.parse(bruto);
    } catch (e) {
      throw new Error(`resposta inesperada ao enviar o arquivo: ${bruto.slice(0, 160)}`);
    }
    if (json.error) throw new Error(json.error);
    if (!json.url) throw new Error('o servidor aceitou o envio mas não devolveu o endereço do arquivo');
    return json;
  }

  function escapaAtributo(s) {
    return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
  }

  // Mesmo formato do rótulo que já existe nas suas salas: a imagem centralizada,
  // ocupando a largura do tópico, e o texto (se houver) logo abaixo.
  function htmlDoRotulo(urlImagem, nomeImagem, textoAbaixo) {
    const src = escapaAtributo(urlImagem);
    const alt = escapaAtributo(String(nomeImagem || '').replace(/\.[^.]+$/, ''));
    const imagem = `<p style="text-align: center;"><img class="img-fluid" style="width: 100%;" src="${src}" alt="${alt}"></p>`;
    const texto = (textoAbaixo || '').trim();
    return texto ? `${imagem}\n${texto}` : imagem;
  }

  // Copia campo a campo o formulário que veio do DOMParser (new FormData(form)
  // não funciona em documento que não está na tela).
  function camposDoFormulario(form) {
    const dados = new FormData();
    form.querySelectorAll('input, select, textarea').forEach((campo) => {
      const tipo = (campo.type || '').toLowerCase();
      if (!campo.name || campo.disabled) return;
      if (['button', 'reset', 'file', 'image', 'submit'].includes(tipo)) return;
      if ((tipo === 'checkbox' || tipo === 'radio') && !campo.checked) return;
      if (campo.tagName === 'SELECT') {
        const escolhidas = [...campo.options].filter((o) => o.selected);
        (escolhidas.length ? escolhidas : campo.options.length ? [campo.options[0]] : [])
          .forEach((o) => dados.append(campo.name, o.value));
        return;
      }
      dados.append(campo.name, campo.value);
    });
    return dados;
  }

  function apertaOBotaoDeSalvar(form, dados) {
    // Pego o botão de salvar pelo nome, nunca o "Cancelar": mandar o cancelar
    // junto faria o Moodle descartar tudo.
    const botoes = [...form.querySelectorAll('input[type="submit"][name], button[type="submit"][name]')];
    const salvar = botoes.find((b) => /^submitbutton/.test(b.name)) || botoes.find((b) => !/cancel/i.test(b.name));
    if (salvar) dados.set(salvar.name, salvar.value || 'Salvar');
    else dados.set('submitbutton', 'Salvar alterações');
  }

  async function enviaFormulario(form, urlDaPagina, dados) {
    const acao = form.getAttribute('action')
      ? new URL(form.getAttribute('action'), urlDaPagina).href
      : urlDaPagina;
    return baixaPaginaComHtml(acao, { method: 'POST', body: dados });
  }

  // Dá nome à seção recém-criada. É esse nome que marca a sala como já feita.
  async function nomeiaSecao(secaoId, nomeDaSecao) {
    const url = `${ORIGEM}/course/editsection.php?id=${encodeURIComponent(secaoId)}&sr=0`;
    const pagina = await baixaPaginaComHtml(url);
    if (!pagina.doc) throw new Error(`não consegui abrir a tela de edição da seção (${pagina.erro || pagina.status})`);

    const form = pagina.doc.querySelector('form[action*="editsection.php"]')
      || pagina.doc.querySelector('#region-main form');
    if (!form) throw new Error('não achei o formulário de edição da seção');

    const dados = camposDoFormulario(form);
    dados.set('name', nomeDaSecao);
    // Versões mais antigas do Moodle têm a caixa "usar nome padrão": com ela
    // marcada o nome digitado seria ignorado.
    form.querySelectorAll('input[name*="usedefaultname"]').forEach((c) => dados.set(c.name, '0'));
    apertaOBotaoDeSalvar(form, dados);

    const resposta = await enviaFormulario(form, url, dados);
    if (!resposta.ok) throw new Error(`o Moodle recusou o nome da seção (HTTP ${resposta.status})`);
    if (/editsection\.php/.test(resposta.url) && !/\/course\/view\.php/.test(resposta.url)) {
      const erro = txt(resposta.doc && resposta.doc.querySelector('.error, .alert-danger, .invalid-feedback'));
      throw new Error(erro || 'a tela de edição da seção voltou sem salvar o nome');
    }
  }

  // Cria dentro da seção um rótulo ("Área de texto e mídia") com a imagem —
  // o mesmo tipo de item que já é usado hoje nas salas.
  async function criaRotuloComImagem(courseid, secaoId, imagem, nomeDoRotulo, textoAbaixo, anota) {
    const url = `${ORIGEM}/course/modedit.php?add=label&type=&course=${encodeURIComponent(courseid)}`
      + `&sectionid=${encodeURIComponent(secaoId)}&return=0&sr=0`;
    const pagina = await baixaPaginaComHtml(url);
    if (!pagina.doc) throw new Error(`não consegui abrir a tela do rótulo (${pagina.erro || pagina.status})`);

    const form = pagina.doc.querySelector('form[action*="modedit.php"]')
      || pagina.doc.querySelector('#region-main form');
    if (!form) throw new Error('não achei o formulário do rótulo');

    let urlImagem = imagem.url;
    let nomeImagem = imagem.url;

    if (imagem.arquivo) {
      const campoItemid = form.querySelector('input[name="introeditor[itemid]"]');
      if (!campoItemid || !campoItemid.value) throw new Error('não achei a área de rascunho do texto do rótulo');
      const ctxId = achaContextId(pagina.html) || '';
      const repoId = await idDoRepositorioDeEnvio(pagina.html, ctxId, anota);
      const enviado = await enviaArquivoParaRascunho(imagem.arquivo, campoItemid.value, repoId, ctxId);
      urlImagem = enviado.url;
      nomeImagem = enviado.file || imagem.arquivo.name;
    }

    const dados = camposDoFormulario(form);
    dados.set('name', nomeDoRotulo);
    dados.set('introeditor[text]', htmlDoRotulo(urlImagem, nomeImagem, textoAbaixo));
    dados.set('introeditor[format]', '1');
    apertaOBotaoDeSalvar(form, dados);

    const resposta = await enviaFormulario(form, url, dados);
    if (!resposta.ok) throw new Error(`o Moodle recusou o rótulo (HTTP ${resposta.status})`);
    if (/modedit\.php/.test(resposta.url) && !/\/course\/(view|section)\.php/.test(resposta.url)) {
      const erro = txt(resposta.doc && resposta.doc.querySelector('.error, .alert-danger, .invalid-feedback'));
      throw new Error(erro || 'o formulário do rótulo voltou sem salvar');
    }
  }

  // ----------------------------------------------------------------------
  // Uma sala
  // ----------------------------------------------------------------------
  async function tratarSala(curso, imagem, soConferir, nomeDaSecao, textoAbaixo, anota) {
    const estado = await pegaEstadoCurso(curso.id);

    const jaTem = secaoJaExistente(estado, nomeDaSecao);
    if (jaTem) return { situacao: 'pulada', detalhe: `já tem a seção "${nomeCruDaSecao(jaTem)}"` };

    if (soConferir) return { situacao: 'receberia', detalhe: 'ainda não tem a seção' };

    const { secao } = await criaSecaoNoTopo(curso.id, estado);
    await dorme(CONFIG_IMAGEM.pausaAjaxMs);
    await nomeiaSecao(secao.id, nomeDaSecao);
    await dorme(CONFIG_IMAGEM.pausaAjaxMs);
    await criaRotuloComImagem(curso.id, secao.id, imagem, nomeDaSecao, textoAbaixo, anota);
    return { situacao: 'feita', detalhe: `seção "${nomeDaSecao}" criada no topo com a imagem dentro` };
  }

  // ----------------------------------------------------------------------
  // Execução do "adicionar imagem"
  // ----------------------------------------------------------------------
  let resumoImagem = '';

  function textoResumoImagem(resultados, soConferir, nomeDaSecao, marcadas) {
    const conta = (situacao) => resultados.filter((r) => r.situacao === situacao).length;
    const linhas = [];
    linhas.push(soConferir ? '=== CONFERÊNCIA (nada foi alterado) ===' : '=== IMAGEM NO TOPO DAS SALAS ===');
    linhas.push(`Seção: ${nomeDaSecao}`);
    linhas.push(`Cursos (categorias): ${marcadas.map((c) => c.nome).join(', ')}`);
    linhas.push(`Salas analisadas: ${resultados.length}`);
    if (soConferir) linhas.push(`Receberiam a imagem: ${conta('receberia')}`);
    else linhas.push(`Com a imagem adicionada: ${conta('feita')}`);
    linhas.push(`Puladas: ${conta('pulada')}`);
    linhas.push(`Com erro: ${conta('erro')}`);

    const porSituacao = [
      ['receberia', 'VÃO RECEBER A IMAGEM'],
      ['feita', 'IMAGEM ADICIONADA'],
      ['pulada', 'PULADAS'],
      ['erro', 'COM ERRO'],
    ];
    porSituacao.forEach(([chave, rotulo]) => {
      const desta = resultados.filter((r) => r.situacao === chave);
      if (!desta.length) return;
      linhas.push('');
      linhas.push(`${rotulo} (${desta.length}):`);
      desta.forEach((r) => {
        const extra = chave === 'receberia' ? '' : ` — ${r.detalhe}`;
        linhas.push(`  - [${r.categoria}] ${r.curso}${extra}`);
      });
    });
    return linhas.join('\n');
  }

  function copiarResumoImagem() {
    navigator.clipboard.writeText(resumoImagem)
      .then(() => log('Resumo copiado.', 'ok'))
      .catch(() => log('Não consegui copiar. Selecione o texto do log e copie à mão.', 'erro'));
  }

  async function rodaImagem(soConferir) {
    if (rodando) { log('Espere a execução que já está em andamento terminar.', 'aviso'); return; }

    const contexto = contextoDaCategoria();
    if (!contexto.ok) { log(contexto.motivo, 'erro'); return; }

    const nomeDaSecao = (campoNomeSecao.value || '').trim() || CONFIG_IMAGEM.nomeDaSecao;
    const textoAbaixo = campoTextoAbaixo.value || CONFIG_IMAGEM.textoAbaixoDaImagem;

    // A imagem pode vir de um arquivo do computador ou de um endereço que já
    // esteja publicado (como o calendário que vocês usam hoje).
    const arquivo = campoArquivo.files && campoArquivo.files[0];
    const endereco = (campoUrlImagem.value || '').trim();
    const imagem = arquivo ? { arquivo } : (endereco ? { url: endereco } : null);
    if (!soConferir && !imagem) {
      log('Escolha uma imagem do computador ou cole o endere\u00e7o de uma imagem.', 'erro');
      return;
    }

    const marcadas = [...listaCategorias.querySelectorAll('input[type="checkbox"]:checked')]
      .map((c) => ({ id: c.value, nome: c.dataset.nome || c.value }));
    if (!marcadas.length) { log('Marque pelo menos um curso (categoria) na lista.', 'erro'); return; }

    if (!soConferir) {
      const comoChamar = arquivo ? arquivo.name : endereco;
      const aviso = `Adicionar a imagem "${comoChamar}" numa seção nova chamada "${nomeDaSecao}", `
        + `no topo de TODAS as salas de: ${marcadas.map((c) => c.nome).join(', ')}.\n\n`
        + 'Salas que já tiverem essa seção serão puladas. Confirmar?';
      if (!window.confirm(aviso)) { log('Cancelado por você.', 'aviso'); return; }
    }

    rodando = true; interromper = false;
    botaoParar.style.display = ''; botaoCopiarImagem.style.display = 'none';
    botaoConferirImagem.disabled = true; botaoAplicarImagem.disabled = true; botao.disabled = true;
    painelLog.innerHTML = '';

    const resultados = [];
    try {
      log(soConferir
        ? `Conferindo (nada será alterado) — seção procurada: "${nomeDaSecao}".`
        : `Adicionando a imagem "${arquivo ? arquivo.name : endereco}" — seção "${nomeDaSecao}".`);

      for (const categoria of marcadas) {
        if (interromper) break;
        log(`Categoria: ${categoria.nome}`);
        const salas = await salasDaCategoria(categoria.id, contexto, (m) => log(m, 'aviso'));
        log(`   ${salas.length} sala(s) encontrada(s).`);

        for (const curso of salas) {
          if (interromper) { log('Interrompido por você.', 'aviso'); break; }
          const curto = nomeCurto(curso);

          if (!soConferir && perguntarSalaASala.checked) {
            if (!window.confirm(`Adicionar a imagem em "${curto}"?`)) {
              resultados.push({ categoria: categoria.nome, curso: curto, situacao: 'pulada', detalhe: 'você pulou esta sala' });
              log(`   ${curto}: pulada por você.`, 'aviso');
              continue;
            }
          }

          try {
            const r = await tratarSala(curso, imagem, soConferir, nomeDaSecao, textoAbaixo, (m) => log(m));
            resultados.push(Object.assign({ categoria: categoria.nome, curso: curto }, r));
            const cor = r.situacao === 'feita' ? 'ok' : (r.situacao === 'pulada' ? 'aviso' : undefined);
            log(`   ${curto}: ${r.detalhe}`, cor);
          } catch (e) {
            resultados.push({ categoria: categoria.nome, curso: curto, situacao: 'erro', detalhe: e.message });
            log(`   ${curto}: ERRO — ${e.message}`, 'erro');
          }
          await dorme(CONFIG_IMAGEM.pausaEntreSalasMs);
        }
      }

      resumoImagem = textoResumoImagem(resultados, soConferir, nomeDaSecao, marcadas);
      log('');
      resumoImagem.split('\n').forEach((l) => log(l));
      if (resultados.length) botaoCopiarImagem.style.display = '';
    } catch (e) {
      log(`Parou: ${e.message}`, 'erro');
    } finally {
      rodando = false; interromper = false;
      botaoParar.style.display = 'none'; botaoParar.textContent = 'Parar';
      botaoConferirImagem.disabled = false; botaoAplicarImagem.disabled = false; botao.disabled = false;
    }
  }
