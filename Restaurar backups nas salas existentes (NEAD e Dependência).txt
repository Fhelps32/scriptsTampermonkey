// ==UserScript==
// @name         Restaurar backups nas salas existentes (NEAD e Dependência)
// @namespace    https://nead.ugb.edu.br/
// @version      1.0
// @description  Onde usar: abra a categoria com a lista das salas do semestre NOVO e use o painel no canto da tela. Para cada sala: apaga o conteúdo que está lá e põe no lugar o conteúdo do backup do semestre anterior. Os alunos matriculados NÃO são mexidos.
// @author       Você
// @match        https://nead.ugb.edu.br/course/index.php*
// @match        https://dependencia.ugb.edu.br/course/index.php*
// @icon         https://nead.ugb.edu.br/favicon.ico
// @grant        none
// @run-at       document-idle
// ==/UserScript==

(function () {
  'use strict';

  // ----------------------------------------------------------------------
  // Ajustes
  // ----------------------------------------------------------------------
  const CONFIG = {
    // O que fazer com o que já está na sala de destino:
    //   'apagar'  — apaga o conteúdo da sala e põe o do backup no lugar
    //   'mesclar' — soma o conteúdo do backup ao que já existe
    modoDeRestauracao: 'apagar',

    // Trava de segurança, explicada no bloco sobre matrículas mais abaixo:
    // sem o ajuste que preserva matrículas, a restauração nem chega a rodar.
    exigirManterMatriculas: true,
    conferirMatriculasDepois: true,

    preservarConfiguracoes: true, // não deixar o backup sobrescrever nome e datas da sala

    // Conteúdo da sala de destino — as mesmas regras do script "Conferir
    // material de apoio", para "sala com conteúdo" querer dizer a mesma coisa
    // em todos: fórum não conta, rótulo não conta e item oculto não conta.
    rotulosContamComoConteudo: false,
    linksExternosContamComoConteudo: true, // rótulo com link para fora do Moodle conta
    tiposIgnorados: ['forum'],

    maxEtapas: 12,                // trava: a restauração tem 7 telas, 12 é folga
    repoIdUpload: null,           // preencha só se o script não achar sozinho
    pausaMs: 2000,                // a restauração roda no servidor: espaço entre as salas
    usarIframe: true,
    esperaIframeMs: 12000,
  };

  // Os dois destinos "nesta sala" do Moodle. Nenhum dos dois cria sala nova.
  const ALVO_APAGAR = '0';   // apaga o conteúdo da sala e restaura por cima
  const ALVO_MESCLAR = '1';  // soma ao que já existe

  const alvoEscolhido = () => (CONFIG.modoDeRestauracao === 'mesclar' ? ALVO_MESCLAR : ALVO_APAGAR);

  // ----------------------------------------------------------------------
  // Matrículas: por que existe uma trava aqui
  // ----------------------------------------------------------------------
  // Apagar o conteúdo do curso, no Moodle, passa pela mesma função que apaga
  // as matrículas. Quem decide é um ajuste do assistente:
  //
  //   restore_controller: $options['keep_roles_and_enrolments'] =
  //                           $this->get_setting_value('keep_roles_and_enrolments');
  //   remove_course_contents: // Remove roles and enrolments by default.
  //                           if (empty($options['keep_roles_and_enrolments'])) {
  //                               enrol_course_delete($course, $userid);
  //                               role_unassign_all(...);
  //
  // Ou seja: com esse ajuste em "Não", apagar o conteúdo leva os alunos junto.
  //
  // O ajuste é criado em restore_course_task, então é do CURSO e não da raiz:
  // o campo se chama setting_course_keep_roles_and_enrolments, é um select
  // Sim/Não com padrão NÃO, e aparece na tela de ajustes do curso — a mesma de
  // "Sobrescrever a configuração do curso", que vem DEPOIS da tela de ajustes
  // gerais. Ele só existe quando o destino escolhido apaga o conteúdo.
  //
  // Por isso o script marca o ajuste, confere que conseguiu marcar antes de
  // deixar a restauração rodar, e ainda compara a quantidade de matriculados
  // antes e depois.
  const RE_AJUSTE_DE_MATRICULAS = /keep_roles_and_enrolments/i;
  const RE_AJUSTE_DE_GRUPOS = /keep_groups_and_groupings/i;

  const ORIGEM = location.origin;
  const abs = (u) => new URL(u, ORIGEM).href;
  const dorme = (ms) => new Promise((r) => setTimeout(r, ms));
  const txt = (el) => (el && el.textContent || '').replace(/ /g, ' ').replace(/\s+/g, ' ').trim();
  const nomeCurto = (nome) => nome.replace(/\[[^\]]*\]\s*/g, '').replace(/\s*[•⬩].*$/, '').trim();
  const norm = (s) => s.normalize('NFKD').replace(/[\u0300-\u036f]/g, '').toUpperCase().replace(/\s+/g, ' ').trim();

  // Mesma chave do "Importar bancos de questões": tira [26.2], (VR-ADM.N), acentos
  // e símbolos, de modo que o nome da matéria case entre um semestre e outro.
  const chave = (s) =>
    s.replace(/\[[^\]]*\]/g, ' ')
      .replace(/\([^)]*\)/g, ' ')
      .normalize('NFD').replace(/[̀-ͯ]/g, '')
      .toUpperCase().replace(/[^A-Z0-9]/g, '');

  // ----------------------------------------------------------------------
  // Rede
  // ----------------------------------------------------------------------
  async function baixaPagina(url) {
    try {
      const resp = await fetch(url, { credentials: 'same-origin' });
      const html = await resp.text();
      return { url: resp.url || url, status: resp.status, ok: resp.ok, html, doc: new DOMParser().parseFromString(html, 'text/html') };
    } catch (e) {
      return { url, status: 0, ok: false, html: '', doc: null, erro: e.message };
    }
  }

  function carregaComJs(url, temOQuePreciso) {
    return new Promise((resolve) => {
      const frame = document.createElement('iframe');
      frame.style.cssText = 'position:fixed;left:-10000px;top:0;width:1280px;height:900px;border:0;';
      frame.src = url;
      document.body.appendChild(frame);
      const limite = Date.now() + CONFIG.esperaIframeMs;
      const timer = setInterval(() => {
        let doc = null;
        try { doc = frame.contentDocument; } catch (e) { /* ignora */ }
        if (doc && doc.readyState === 'complete' && temOQuePreciso(doc)) {
          clearInterval(timer);
          const copia = doc.cloneNode(true);
          setTimeout(() => frame.remove(), 100);
          resolve(copia);
          return;
        }
        if (Date.now() > limite) {
          clearInterval(timer);
          const copia = doc ? doc.cloneNode(true) : null;
          frame.remove();
          resolve(copia);
        }
      }, 400);
    });
  }

  // ----------------------------------------------------------------------
  // Salas de destino (a categoria aberta)
  // ----------------------------------------------------------------------
  async function listarCursos() {
    const paginas = new Map([[location.href, document]]);
    document.querySelectorAll('.paging a[href], nav a[href*="page="]').forEach((a) => {
      if (!paginas.has(a.href)) paginas.set(a.href, null);
    });

    const cursos = new Map();
    for (const [url, cache] of paginas) {
      let doc = cache;
      if (!doc) { const r = await baixaPagina(url); if (!r.doc) continue; doc = r.doc; }
      const escopo = doc.querySelector('#region-main') || doc.body;
      escopo.querySelectorAll('a[href*="/course/view.php?id="]').forEach((a) => {
        const id = new URL(a.getAttribute('href'), ORIGEM).searchParams.get('id');
        const nome = txt(a);
        if (id && nome && !cursos.has(id)) cursos.set(id, nome);
      });
    }
    return [...cursos].map(([id, nome]) => ({ id, nome }));
  }


  // Um rótulo é texto solto e, por si, não é material de estudo. Mas quando ele
  // traz link para fora do Moodle — Drive, YouTube, um PDF em outro site —, é
  // ali que o material está: nesse caso o item conta como conteúdo.
  //
  // O mesmo vale para link que está na seção mas fora de qualquer atividade,
  // como no resumo/descrição da seção: para o aluno, é material do mesmo jeito.
  //
  // "Fora do Moodle" é decidido pelo endereço. Isso importa: a própria seção
  // tem links de menu, permalink e edição, todos do site — se qualquer link
  // contasse, toda seção passaria a ter conteúdo e a conferência perderia o
  // sentido.
  const HOSPEDEIRO = location.host;

  function ehEnderecoExterno(href) {
    if (!href || /^(#|mailto:|tel:|javascript:)/i.test(href)) return false;
    let alvo;
    try { alvo = new URL(href, ORIGEM); } catch (e) { return false; }
    if (!/^https?:$/.test(alvo.protocol)) return false;
    return !!alvo.host && alvo.host !== HOSPEDEIRO;
  }

  function linksExternosEm(no) {
    const achados = new Set();
    if (!no) return achados;
    no.querySelectorAll('a[href]').forEach((a) => {
      const href = a.getAttribute('href') || '';
      if (ehEnderecoExterno(href)) achados.add(new URL(href, ORIGEM).href);
    });
    // Endereço escrito no texto, sem virar link.
    const texto = txt(no) || '';
    (texto.match(/https?:\/\/[^\s<>"')]+/gi) || []).forEach((u) => {
      if (ehEnderecoExterno(u)) achados.add(u);
    });
    return achados;
  }

  const temLinkExterno = (item) => linksExternosEm(item).size > 0;

  // Links da seção que não estão dentro de nenhuma atividade (resumo da seção,
  // por exemplo). As atividades saem do clone porque já foram contadas uma a
  // uma, e os controles da seção saem para não entrarem como material.
  function linksExternosSoltos(secao) {
    const clone = secao.cloneNode(true);
    clone.querySelectorAll('li.activity, li.course-module, [data-for="cmitem"], .activity-item')
      .forEach((n) => n.remove());
    clone.querySelectorAll('.controlmenu, .menu, .dropdown, .action-menu, button, [role="button"], ' +
      '.sr-only, .accesshide, .visually-hidden, .section_action_menu, .sectionbadges')
      .forEach((n) => n.remove());
    return [...linksExternosEm(clone)];
  }

  // Quanto a sala de destino já tem. Serve de aviso: mesclar soma o backup ao
  // que já está lá, então uma sala com conteúdo pode acabar com tudo em dobro.
  //
  // A contagem é a mesma do "Conferir material de apoio": fórum não é
  // material de estudo, rótulo é texto solto e item oculto não está valendo
  // para o aluno — nenhum dos três conta como conteúdo.
  function contaConteudo(doc) {
    if (!doc) return { secoes: 0, itens: 0 };
    const nos = new Set();
    doc.querySelectorAll('li.section, li.course-section, [data-for="section"], .course-section').forEach((n) => nos.add(n));
    const secoes = [...nos].filter((n) => ![...nos].some((o) => o !== n && o.contains(n)));

    let itens = 0, ocultos = 0, ignorados = 0;
    for (const secao of secoes) {
      const candidatos = new Set();
      secao.querySelectorAll('li.activity, li.course-module, [data-for="cmitem"], .activity-item')
        .forEach((n) => candidatos.add(n));
      const atividades = [...candidatos].filter((n) => ![...candidatos].some((o) => o !== n && o.contains(n)));

      for (const item of atividades) {
        const classe = item.className || '';
        const mtipo = (classe.match(/modtype_(\w+)/) || [])[1] || '';
        const ehRotulo = /modtype_label|\blabel\b/.test(classe);
        const escondido = /\bhidden\b|dimmed/.test(classe) ||
          /OCULTO|N[ÃA]O DISPON[ÍI]VEL|HIDDEN FROM STUDENTS/.test(norm(txt(item.querySelector('.badge, .availabilityinfo')) || ''));

        const porLinkExterno = CONFIG.linksExternosContamComoConteudo && ehRotulo && temLinkExterno(item);
        if (CONFIG.tiposIgnorados.includes(mtipo.toLowerCase())) { ignorados++; continue; }
        if (ehRotulo && !CONFIG.rotulosContamComoConteudo && !porLinkExterno) continue;
        if (escondido) { ocultos++; continue; }
        itens++;
      }

      // Links fora das atividades (resumo da seção) também são material.
      if (CONFIG.linksExternosContamComoConteudo) itens += linksExternosSoltos(secao).length;
    }
    return { secoes: secoes.length, itens, ocultos, ignorados };
  }

  // O id de contexto da sala: é ele que a página de restauração pede. Vem do
  // link "Reutilizar curso" e, se ele não estiver no HTML, da configuração que
  // o Moodle injeta na própria página.
  async function contextoDoCurso(courseid) {
    const r = await baixaPagina(`${ORIGEM}/course/view.php?id=${courseid}`);
    const link = r.doc ? r.doc.querySelector('a[href*="restorefile.php"], a[href*="backup/restorefile"]') : null;
    if (link) {
      const ctx = new URL(link.getAttribute('href'), ORIGEM).searchParams.get('contextid');
      if (ctx) return ctx;
    }
    const m = (r.html || '').match(/"contextid"\s*:\s*(\d+)/) || (r.html || '').match(/"contextInstanceId"\s*:\s*(\d+)/);
    return m ? m[1] : null;
  }

  // ----------------------------------------------------------------------
  // Arquivos escolhidos
  // ----------------------------------------------------------------------
  let arquivosPorChave = new Map();

  // O nome pode vir do jeito que o "Fazer backup das salas" baixa (o nome da sala) ou
  // com o padrão do Moodle: backup-moodle2-course-123-abreviatura-20260911-1730-nu.mbz
  function chaveDoArquivo(nome) {
    const base = nome
      .replace(/\.mbz$/i, '')
      .replace(/^backup-moodle2-course-\d+-/i, '')
      .replace(/-\d{8}-\d{4}(-nu)?$/i, '');
    return chave(base);
  }

  function registraArquivos(lista) {
    arquivosPorChave = new Map();
    let repetidos = 0;
    for (const arq of lista) {
      const k = chaveDoArquivo(arq.name);
      if (!k) continue;
      if (arquivosPorChave.has(k)) { repetidos++; continue; }
      arquivosPorChave.set(k, arq);
    }
    log(`${lista.length} arquivo(s) escolhido(s): ${arquivosPorChave.size} matéria(s) reconhecida(s)` +
        (repetidos ? `, ${repetidos} com nome repetido (ignorados)` : ''));
  }

  // ----------------------------------------------------------------------
  // Envio do .mbz para a área de rascunho
  // ----------------------------------------------------------------------
  // O id do repositório "Enviar um arquivo" aparece no JSON que inicializa o
  // seletor de arquivos. A ordem das chaves muda entre versões, então tento
  // os dois formatos antes de ler o bloco inteiro.
  function extraiRepoDoHtml(html) {
    const variantes = [
      /"id"\s*:\s*"?(\d+)"?[^{}]{0,600}?"type"\s*:\s*"upload"/,
      /"type"\s*:\s*"upload"[^{}]{0,600}?"id"\s*:\s*"?(\d+)"?/,
    ];
    for (const re of variantes) { const m = html.match(re); if (m) return m[1]; }
    return null;
  }

  function inspecionaComJs(url, extrai) {
    return new Promise((resolve) => {
      const frame = document.createElement('iframe');
      frame.style.cssText = 'position:fixed;left:-10000px;top:0;width:1280px;height:900px;border:0;';
      frame.src = url;
      document.body.appendChild(frame);
      const limite = Date.now() + CONFIG.esperaIframeMs;
      const timer = setInterval(() => {
        let achado = null;
        try {
          const w = frame.contentWindow, d = frame.contentDocument;
          if (d && d.readyState === 'complete') achado = extrai(w, d);
        } catch (e) { /* ignora */ }
        if (achado != null || Date.now() > limite) {
          clearInterval(timer);
          frame.remove();
          resolve(achado);
        }
      }, 500);
    });
  }

  let repoUploadCache = null;

  async function descobreRepoUpload(urlPagina, html) {
    if (CONFIG.repoIdUpload) return CONFIG.repoIdUpload;
    if (repoUploadCache) return repoUploadCache;

    // Quase sempre o id já está no HTML que acabei de buscar: só caio no
    // iframe (que custa segundos) quando não está.
    const noHtml = extraiRepoDoHtml(html || '');
    if (noHtml) { repoUploadCache = noHtml; return noHtml; }

    const achado = await inspecionaComJs(urlPagina, (w, d) => {
      const instancias = w.M && w.M.core_filepicker && w.M.core_filepicker.instances;
      for (const k in (instancias || {})) {
        const reps = instancias[k] && instancias[k].options && instancias[k].options.repositories;
        for (const r in (reps || {})) {
          if (reps[r] && reps[r].type === 'upload') return String(reps[r].id || r);
        }
      }
      return extraiRepoDoHtml(d.documentElement.innerHTML);
    });
    if (achado) repoUploadCache = achado;
    return achado;
  }

  // O campo do seletor de arquivos guarda o id da área de rascunho. O nome do
  // campo muda entre versões, então procuro pelo formato do valor e dou
  // preferência a quem tem "file"/"backup" no nome.
  const RE_CAMPOS_DE_CONTROLE = /^(sesskey|_qf__|mform_isexpanded|contextid|courseid|id|MAX_FILE_SIZE|submitbutton|cancel)/i;

  function campoDoArquivo(form) {
    const candidatos = [...form.querySelectorAll('input[type="hidden"][name]')]
      .filter((i) => !RE_CAMPOS_DE_CONTROLE.test(i.name) && /^\d{3,}$/.test(i.value || ''));
    return candidatos.find((i) => /file|backup/i.test(i.name)) || candidatos[0] || null;
  }

  // O seletor de arquivos do Moodle deixa marcas no HTML. Procurar por elas é
  // mais seguro do que aceitar "qualquer formulário com um campo escondido
  // numérico", que casa com formulário de busca e manda o arquivo para o lugar
  // errado — e aí a tela volta para o começo sem dizer por quê.
  const SELETOR_DE_ARQUIVO =
    '[data-fieldtype="filemanager"], [data-fieldtype="filepicker"], .filemanager, .filepicker, ' +
    '.ffilemanager, .ffilepicker, input[name="MAX_FILE_SIZE"]';

  // A página de restauração traz outros formulários da barra do curso, e um
  // deles — o de modo de edição — tem um campo "context" com número grande,
  // igualzinho a um id de rascunho. Por isso o formulário de envio só é aceito
  // quando tem o seletor de arquivos do Moodle ou quando o endereço dele é o
  // da própria página de restauração.
  function formularioDeEnvio(doc) {
    if (!doc) return null;
    const formularios = [...doc.querySelectorAll('form')]
      .filter((f) => campoDoArquivo(f) && botaoDeAvancar(f));
    return formularios.find((f) => f.querySelector(SELETOR_DE_ARQUIVO))
        || formularios.find((f) => /restorefile\.php/.test(f.getAttribute('action') || ''))
        || null;
  }

  async function enviaParaRascunho(arquivo, dados) {
    const fd = new FormData();
    fd.set('repo_upload_file', arquivo, arquivo.name);
    fd.set('itemid', dados.itemid);
    fd.set('repo_id', dados.repoId);
    fd.set('ctx_id', dados.ctxId || '');
    fd.set('env', 'filepicker');
    fd.set('sesskey', dados.sesskey);
    fd.set('title', arquivo.name);
    fd.set('savepath', '/');
    fd.set('author', '');
    fd.set('overwrite', 1);

    const resp = await fetch(`${ORIGEM}/repository/repository_ajax.php?action=upload`, {
      method: 'POST', credentials: 'same-origin', body: fd,
    });
    const texto = await resp.text();
    let json;
    try { json = JSON.parse(texto); } catch (e) {
      throw new Error(`o envio do arquivo devolveu resposta inesperada (HTTP ${resp.status})`);
    }
    if (json.error) throw new Error(`o Moodle recusou o arquivo: ${json.error}`);
    return json;
  }

  // ----------------------------------------------------------------------
  // Assistente de restauração
  // ----------------------------------------------------------------------
  const RE_BOTAO_DE_VOLTA = /cancel|previous|anterior|cancelar/i;
  const RE_SUCESSO = /(restaurad[oa] com sucesso|restored successfully|o curso foi restaurado)/i;
  const RE_ASSINCRONO = /(ass[ií]ncron|em segundo plano|in progress|será processad|sera processad|fila de execu)/i;

  // Ajustes que trariam dados de aluno de volta. O backup vem sem usuários,
  // mas mando zero de qualquer modo: matrícula da sala nova não se mexe.
  const AJUSTES_DE_USUARIO = [
    'setting_root_users',
    'setting_root_role_assignments',
    'setting_root_userscompletion',
    'setting_root_logs',
    'setting_root_grade_histories',
    'setting_root_comments',
  ];

  function botaoDeAvancar(form) {
    const botoes = [...form.querySelectorAll('input[type="submit"], button[type="submit"], button:not([type])')];
    return botoes.find((b) => !RE_BOTAO_DE_VOLTA.test(b.name || '') && !RE_BOTAO_DE_VOLTA.test(txt(b) || b.value || '')) || null;
  }

  // "restorefile.php" não contém "restore.php", então o endereço separa as
  // telas do assistente da tela de envio sem ambiguidade.
  function formularioDoAssistente(doc) {
    if (!doc) return null;
    return [...doc.querySelectorAll('form')].find((f) => {
      if (!botaoDeAvancar(f)) return false;
      if (f.querySelector('input[name="restore"], input[name="stage"]')) return true;
      // Em parte das versões a etapa vai na URL do formulário, não num campo.
      return /backup\/restore\.php/.test(f.getAttribute('action') || '');
    }) || null;
  }

  // Quando uma tela não é reconhecida, é isto que transforma "parou na etapa 1"
  // em algo que dá para agir: o título, o aviso do Moodle e os formulários.
  function descreveTela(doc) {
    if (!doc) return 'a resposta veio vazia';
    const titulo = txt(doc.querySelector('#region-main h1, #region-main h2, h1')) || '(sem título)';
    const erro = mensagemDeErro(doc);
    const formularios = [...doc.querySelectorAll('form')].map((f) => {
      const acao = (f.getAttribute('action') || '(mesma página)').replace(ORIGEM, '');
      const campos = [...new Set([...f.querySelectorAll('input[name], select[name]')].map((i) => i.name))];
      const botao = botaoDeAvancar(f);
      return `${acao} campos[${campos.slice(0, 10).join(', ')}] ${botao ? `botão "${txt(botao) || botao.value}"` : 'sem botão'}`;
    });
    return `tela "${titulo}"${erro ? ` · aviso do Moodle: ${erro}` : ''} · ` +
      (formularios.length ? `formulários: ${formularios.join('  ||  ')}` : 'nenhum formulário');
  }

  // A tela de destino traz três formulários: novo curso, esta sala e buscar
  // outra sala. Pego pelo valor do alvo, não pela ordem nem pelo texto: só o
  // bloco "restaurar nesta sala" tem a opção de mesclar com valor 1.
  // Os três blocos da tela de destino carregam "target" e "targetid", mas dois
  // deles são caixas de busca ("Categorias de pesquisa", "Buscar cursos") que
  // não levam a lugar nenhum. O bloco que vale é o que tem a opção de mesclar
  // como escolha de verdade e um botão de seguir adiante.
  const RE_BOTAO_DE_BUSCA = /search|busca|pesquis/i;

  function ehFormularioDeBusca(form) {
    const botao = botaoDeAvancar(form);
    if (!botao) return true;
    if (RE_BOTAO_DE_BUSCA.test(botao.name || '') || RE_BOTAO_DE_BUSCA.test(txt(botao) || botao.value || '')) return true;
    return !!form.querySelector('input[name="catsearch"], input[name="search"]');
  }

  function formularioDeDestino(doc) {
    if (!doc) return null;
    const alvo = alvoEscolhido();
    const candidatos = [...doc.querySelectorAll('form')].filter((f) =>
      f.querySelector(`input[name="target"][value="${alvo}"]`) && !ehFormularioDeBusca(f));
    const comEscolha = candidatos.find((f) =>
      (f.querySelector(`input[name="target"][value="${alvo}"]`).type || '').toLowerCase() === 'radio');
    return comEscolha || candidatos[0] || null;
  }

  function terminou(doc) {
    if (!doc) return false;
    const texto = txt(doc.querySelector('#region-main') || doc.body);
    if (RE_SUCESSO.test(texto)) return true;
    return !formularioDoAssistente(doc) && !formularioDeDestino(doc) && !formularioDeEnvio(doc);
  }

  function mensagemDeErro(doc) {
    if (!doc) return 'resposta vazia do servidor';
    return [...doc.querySelectorAll(
      '.errorbox, .alert-danger, .alert-warning, .invalid-feedback, .form-control-feedback, [id^="id_error_"]')]
      .map((e) => txt(e)).filter(Boolean).slice(0, 3).join(' / ');
  }

  // Serializa o formulário como o navegador faria, preservando o estado de
  // cada controle: reenviar só os campos escondidos desmarcaria as atividades
  // e a restauração entraria vazia.
  function dadosDoFormulario(form) {
    try {
      const dados = new FormData(form);
      if ([...dados.keys()].length) return dados;
    } catch (e) { /* documento do DOMParser: copio campo a campo */ }

    const dados = new FormData();
    form.querySelectorAll('input, select, textarea').forEach((campo) => {
      const tipo = (campo.type || '').toLowerCase();
      if (!campo.name || campo.disabled) return;
      if (['submit', 'button', 'reset', 'file', 'image'].includes(tipo)) return;
      if ((tipo === 'checkbox' || tipo === 'radio') && !campo.checked) return;
      if (campo.tagName === 'SELECT') {
        const escolhidas = [...campo.options].filter((o) => o.selected);
        (escolhidas.length ? escolhidas : campo.options.length ? [campo.options[0]] : [])
          .forEach((o) => dados.append(campo.name, o.value));
        return;
      }
      dados.append(campo.name, campo.value);
    });
    return dados;
  }

  // Devolve { mexeu, viuAjusteDeMatriculas, ehTelaDeAjustes } para quem chama
  // decidir se pode seguir.
  function ajustaDadosDaEtapa(dados, form, anota) {
    const mexeu = [];
    let viuAjusteDeMatriculas = false;

    // O nome do campo muda de prefixo entre versões (setting_root_… /
    // setting_course_…), então procuro pelo miolo do nome.
    for (const campo of form.querySelectorAll('input[name], select[name]')) {
      if (RE_AJUSTE_DE_MATRICULAS.test(campo.name)) {
        dados.set(campo.name, '1');
        viuAjusteDeMatriculas = true;
        if (!mexeu.includes('manter as matrículas e os papéis')) mexeu.push('manter as matrículas e os papéis');
      } else if (RE_AJUSTE_DE_GRUPOS.test(campo.name)) {
        dados.set(campo.name, '1');
        if (!mexeu.includes('manter os grupos')) mexeu.push('manter os grupos');
      }
    }

    for (const nome of AJUSTES_DE_USUARIO) {
      if (form.querySelector(`[name="${nome}"]`)) { dados.set(nome, '0'); mexeu.push(nome.replace('setting_root_', '')); }
    }
    // "Sobrescrever a configuração do curso" traria o nome, o formato e as
    // datas da sala velha por cima da sala nova.
    if (CONFIG.preservarConfiguracoes && form.querySelector('[name="setting_course_overwrite_conf"]')) {
      dados.set('setting_course_overwrite_conf', '0');
      mexeu.push('sem sobrescrever a configuração da sala');
    }
    if (anota && mexeu.length) anota(`   mantendo a sala como está: ${mexeu.join(', ')}`);

    // A tela de ajustes do curso é a que traz os "setting_course_…". É nela
    // que o ajuste de matrículas aparece quando o destino apaga o conteúdo.
    const ehTelaDoCurso = !!form.querySelector('[name^="setting_course_"]');
    return { mexeu, viuAjusteDeMatriculas, ehTelaDoCurso };
  }

  // ----------------------------------------------------------------------
  // Quantos alunos a sala tem (antes e depois)
  // ----------------------------------------------------------------------
  async function contaMatriculados(courseid) {
    const r = await baixaPagina(`${ORIGEM}/user/index.php?id=${courseid}&perpage=5000`);
    if (!r.doc) return null;

    // O Moodle escreve "N participantes encontrados" acima da tabela; quando
    // não escreve, conto as linhas com link para perfil.
    const texto = txt(r.doc.querySelector('#region-main') || r.doc.body);
    const m = texto.match(/(\d+)\s+participantes?\s+encontrad/i) || texto.match(/(\d+)\s+participants?\s+found/i);
    if (m) return Number(m[1]);

    const tabela = r.doc.querySelector('table#participants') || r.doc.querySelector('#participantsform table');
    if (!tabela) return null;
    return [...tabela.querySelectorAll('tbody tr')]
      .filter((tr) => tr.querySelector('a[href*="/user/view.php"], a[href*="/user/profile.php"]')).length;
  }

  async function enviaFormulario(form, urlAtual, ajusta) {
    const dados = dadosDoFormulario(form);
    if (ajusta) ajusta(dados, form);

    const botao = botaoDeAvancar(form);
    if (botao && botao.name) dados.set(botao.name, botao.value || '1');

    const acao = abs(form.getAttribute('action') || urlAtual);
    const resp = await fetch(acao, { method: 'POST', credentials: 'same-origin', body: dados });
    const html = await resp.text();
    return { url: resp.url || acao, doc: new DOMParser().parseFromString(html, 'text/html') };
  }

  const precisaDoAjusteDeMatriculas = () =>
    CONFIG.exigirManterMatriculas && CONFIG.modoDeRestauracao !== 'mesclar';

  // A última tela do assistente é a de revisão, cujo botão executa a
  // restauração — é a partir dela que a sala é alterada.
  const RE_BOTAO_QUE_EXECUTA = /executar|perform|realizar|restaurar agora/i;

  function ehEtapaQueExecuta(form) {
    const botao = botaoDeAvancar(form);
    if (!botao) return false;
    return RE_BOTAO_QUE_EXECUTA.test(txt(botao) || botao.value || '');
  }

  async function restaura(curso, arquivo, anota, pararAntesDeExecutar) {
    let viuAjusteDeMatriculas = false;

    const matriculadosAntes = CONFIG.conferirMatriculasDepois ? await contaMatriculados(curso.id) : null;
    if (matriculadosAntes !== null) anota(`   matriculados antes: ${matriculadosAntes}`);

    const ctxId = await contextoDoCurso(curso.id);
    if (!ctxId) throw new Error('não achei o id de contexto da sala');

    const urlRestauracao = `${ORIGEM}/backup/restorefile.php?contextid=${ctxId}`;
    let r = await baixaPagina(urlRestauracao);
    let doc = r.doc;
    let urlAtual = r.url;

    // 1. Enviar o arquivo para a área de rascunho do formulário
    let form = formularioDeEnvio(doc);
    if (!form && CONFIG.usarIframe) {
      const comJs = await carregaComJs(urlRestauracao, (d) => !!formularioDeEnvio(d));
      if (comJs) { doc = comJs; form = formularioDeEnvio(comJs); }
    }
    if (!form) throw new Error(`não achei o formulário de envio do backup (HTTP ${r.status})`);

    const campo = campoDoArquivo(form);
    if (!campo) throw new Error(`achei o formulário de envio mas não o campo do arquivo — ${descreveTela(doc)}`);
    const sesskey = (form.querySelector('input[name="sesskey"]') || {}).value ||
      ((r.html || '').match(/"sesskey"\s*:\s*"([^"]+)"/) || [])[1];
    const repoId = await descobreRepoUpload(urlRestauracao, r.html);
    if (!repoId) throw new Error('não achei o repositório de envio (preencha repoIdUpload no CONFIG)');

    if (pararAntesDeExecutar) {
      anota(`   envio: campo "${campo.name}" (rascunho ${campo.value}), repositório ${repoId}, sesskey ${sesskey ? 'ok' : 'NÃO ACHEI'}`);
      anota(`   ${descreveTela(doc)}`);
    }

    anota(`   enviando ${arquivo.name}…`);
    const envio = await enviaParaRascunho(arquivo, { itemid: campo.value, repoId, ctxId, sesskey });
    if (pararAntesDeExecutar) anota(`   o Moodle aceitou o arquivo: ${JSON.stringify(envio).slice(0, 200)}`);

    ({ url: urlAtual, doc } = await enviaFormulario(form, urlAtual));
    let erro = mensagemDeErro(doc);
    if (erro) throw new Error(`o Moodle recusou o arquivo enviado: ${erro}`);

    // 2. Percorrer as telas da restauração
    let escolheuDestino = false;
    for (let etapa = 1; etapa <= CONFIG.maxEtapas; etapa++) {
      if (terminou(doc)) {
        const assincrono = RE_ASSINCRONO.test(txt(doc.querySelector('#region-main') || doc.body));
        const matriculadosDepois = matriculadosAntes !== null && !assincrono
          ? await contaMatriculados(curso.id) : null;
        if (matriculadosDepois !== null) {
          anota(`   matriculados depois: ${matriculadosDepois}`);
          if (matriculadosDepois < matriculadosAntes) {
            throw new Error(`ALUNOS PERDIDOS: a sala tinha ${matriculadosAntes} matriculados e ficou com ` +
              `${matriculadosDepois}. Interrompi tudo para não repetir isso nas outras salas.`);
          }
        }
        return { assincrono, matriculadosAntes, matriculadosDepois };
      }

      const destino = formularioDeDestino(doc);
      if (destino) {
        if (pararAntesDeExecutar) anota(`   tela de destino — ${descreveTela(doc)}`);
        anota(CONFIG.modoDeRestauracao === 'mesclar'
          ? '   destino: mesclar com esta sala (soma ao que já existe)'
          : '   destino: apagar o conteúdo desta sala e restaurar por cima');
        ({ url: urlAtual, doc } = await enviaFormulario(destino, urlAtual, (dados) => {
          dados.set('target', alvoEscolhido());
        }));
        escolheuDestino = true;
      } else {
        const atual = formularioDoAssistente(doc);
        if (!atual) {
          // Continuar na tela de envio quer dizer que o arquivo não foi aceito;
          // qualquer outra coisa é uma tela que eu não soube ler.
          throw new Error(formularioDeEnvio(doc)
            ? `o Moodle voltou para a tela de envio do arquivo — ${descreveTela(doc)}`
            : `não reconheci a tela da etapa ${etapa} — ${descreveTela(doc)}`);
        }
        if (pararAntesDeExecutar) anota(`   etapa ${etapa} — ${descreveTela(doc)}`);

        // O diagnóstico percorre as telas e para na última antes da que
        // executa: até aqui o Moodle não apagou nem restaurou nada.
        if (pararAntesDeExecutar && ehEtapaQueExecuta(atual)) {
          anota(`   esta é a tela que executa ("${txt(botaoDeAvancar(atual)) || botaoDeAvancar(atual).value}")`);
          anota(`   ajuste de matrículas encontrado pelo caminho: ${viuAjusteDeMatriculas ? 'SIM' : 'NÃO'}`,
                viuAjusteDeMatriculas ? 'ok' : 'erro');
          anota('   diagnóstico: paro aqui. NADA foi apagado nem restaurado nesta sala.', 'ok');
          return { diagnostico: true };
        }

        // Antes de enviar, olho a tela: se for a de ajustes e o campo que
        // preserva as matrículas não estiver lá, paro aqui. Até este ponto
        // nada foi apagado — quem apaga é a etapa final.
        const previa = ajustaDadosDaEtapa(new FormData(), atual, null);
        if (precisaDoAjusteDeMatriculas() && previa.ehTelaDoCurso && !previa.viuAjusteDeMatriculas) {
          throw new Error('a tela de ajustes do curso veio sem o campo "Manter as funções e inscrições atuais". ' +
            'Apagar o conteúdo sem ele tira os alunos da sala, então parei antes de executar — nada foi apagado. ' +
            'Use modoDeRestauracao: \'mesclar\' ou restaure esta sala pela tela do Moodle.');
        }
        if (previa.viuAjusteDeMatriculas) viuAjusteDeMatriculas = true;

        // A etapa que executa não pode rodar sem essa confirmação.
        if (precisaDoAjusteDeMatriculas() && !viuAjusteDeMatriculas && ehEtapaQueExecuta(atual)) {
          throw new Error('cheguei na etapa que executa a restauração sem ter visto o ajuste que ' +
            'preserva as matrículas. Parei antes de executar — nada foi apagado.');
        }

        ({ url: urlAtual, doc } = await enviaFormulario(atual, urlAtual, (dados, f) => ajustaDadosDaEtapa(dados, f, anota)));
      }

      erro = mensagemDeErro(doc);
      if (erro) throw new Error(`o Moodle recusou na etapa ${etapa}: ${erro}`);
    }

    if (!escolheuDestino) throw new Error('não cheguei na tela de destino — nada foi restaurado');
    throw new Error('a restauração não chegou ao fim dentro do limite de etapas');
  }

      const fundo = document.createElement('div');

        display:flex;align-items:center;justify-content:center;font:13px/1.5 system-ui,sans-serif;`;

      const caixa = document.createElement('div');

      const apagando = CONFIG.modoDeRestauracao !== 'mesclar';

        <div style="background:#eef3ff;border:1px solid #c9d8f5;border-radius:6px;padding:8px;color:#22407e;margin-bottom:12px;">
          Os <strong>alunos matriculados continuam como estão</strong> — o script marca o ajuste que preserva
          matrículas e papéis, e confere a quantidade antes e depois.
          A restauração roda no servidor e demora; deixe a aba aberta.
        </div>
        <label style="display:flex;gap:6px;align-items:center;color:#555;margin-bottom:14px;">
          <input type="checkbox" id="ugb-nao-perguntar"> Restaurar as próximas sem perguntar

      caixa.querySelector('#ugb-restaurar').focus();

      const responde = (res) => {
        if (res === 'restaurar' && caixa.querySelector('#ugb-nao-perguntar').checked) perguntarSempre = false;
        document.removeEventListener('keydown', tecla);
        fundo.remove();
        resolve(res);
      };
      const tecla = (e) => { if (e.key === 'Escape') responde('pular'); };

      caixa.querySelector('#ugb-restaurar').onclick = () => responde('restaurar');

      caixa.querySelector('#ugb-parar').onclick = () => responde('parar');

      document.addEventListener('keydown', tecla);
    });
  }

  let rodando = false, interromper = false;
  let relatorio = [];
  // Guardado fora da execução porque o resumo copiado também precisa dele.
  let sobraramArquivos = [];

  // Vai até a tela de destino e para. Serve para descobrir onde o caminho
  // quebra sem restaurar nada: até essa tela, a sala não é tocada.
  async function diagnosticar() {
    if (rodando) return;
    if (!arquivosPorChave.size) { log('Escolha os arquivos .mbz primeiro.', 'erro'); return; }
    rodando = true; travaBotoes(true); painelLog.innerHTML = '';

    try {
      const cursos = await listarCursos();
      const curso = cursos.find((c) => arquivosPorChave.has(chave(c.nome)));
      if (!curso) { log('Nenhuma sala desta categoria casou com os arquivos escolhidos.', 'erro'); return; }

      const arquivo = arquivosPorChave.get(chave(curso.nome));
      log(`Testando: ${nomeCurto(curso.nome)} (id ${curso.id})`);
      log(`Arquivo: ${arquivo.name}`);

      const ctx = await contextoDoCurso(curso.id);
      log(`Contexto da sala: ${ctx || 'NÃO ACHEI'}`, ctx ? 'ok' : 'erro');
      if (!ctx) return;

      await restaura(curso, arquivo, (m, t) => log(m, t), true);
    } catch (e) {
      log(`Parou aqui: ${e.message}`, 'erro');
      log('Copie o resumo e me mande: é essa descrição da tela que diz o que ajustar.', 'aviso');
    } finally {
      rodando = false; travaBotoes(false);
    }
  }

  async function executar(somenteConferir) {
    if (rodando) return;
    if (!arquivosPorChave.size) { log('Escolha os arquivos .mbz primeiro.', 'erro'); return; }
    rodando = true; interromper = false; perguntarSempre = true; relatorio = [];
    travaBotoes(true); botaoParar.style.display = ''; botaoCopiar.style.display = 'none';
    painelLog.innerHTML = '';

    try {
      const cursos = await listarCursos();
      log(`${cursos.length} salas nesta categoria.`);
      if (!somenteConferir) log('Vou confirmar sala por sala antes de restaurar.');

      const usados = new Set();

      for (let i = 0; i < cursos.length; i++) {
        const curso = cursos[i];
        if (interromper) { log('Interrompido por você.', 'aviso'); break; }

        const curto = nomeCurto(curso.nome);
        const registro = { curso, curto, estado: 'pendente', observacao: '' };
        relatorio.push(registro);
        log(`— ${curto}`);

        const k = chave(curso.nome);
        const arquivo = arquivosPorChave.get(k);
        if (!arquivo) {
          registro.estado = 'sem arquivo';
          registro.observacao = `nenhum .mbz com nome equivalente (chave ${k})`;
          log('   pulada: nenhum arquivo com nome equivalente', 'aviso');
          continue;
        }
        usados.add(k);

        try {
          const r = await baixaPagina(`${ORIGEM}/course/view.php?id=${curso.id}`);
          const contagem = contaConteudo(r.doc);

          if (somenteConferir) {
            registro.estado = 'restauraria';
            registro.observacao = `${arquivo.name}${contagem.itens ? ` — a sala já tem ${contagem.itens} item(ns)` : ''}`;
            log(`   ok: ${arquivo.name} → esta sala` +
                (CONFIG.modoDeRestauracao === 'mesclar'
                  ? `${contagem.itens ? ` (soma aos ${contagem.itens} item(ns) que já estão lá)` : ''}`
                  : `${contagem.itens ? ` (apaga os ${contagem.itens} item(ns) que estão lá)` : ' (sala já vazia)'}`));
            continue;
          }

          const resposta = perguntarSempre
            ? await confirmaSala(curso, arquivo, contagem, i + 1, cursos.length)
            : 'restaurar';
          if (resposta === 'parar') { registro.estado = 'pulada'; log('Interrompido por você.', 'aviso'); break; }
          if (resposta === 'pular') { registro.estado = 'pulada'; log('   pulada por você', 'aviso'); continue; }

          const resultado = await restaura(curso, arquivo, (m) => log(m));
          registro.estado = 'restaurada';
          registro.observacao = arquivo.name;
          if (resultado.matriculadosDepois !== null && resultado.matriculadosDepois !== undefined) {
            registro.observacao += ` — ${resultado.matriculadosDepois} matriculados (antes: ${resultado.matriculadosAntes})`;
          }
          log(`   restaurada: ${arquivo.name}`, 'ok');

          if (resultado.assincrono) {
            registro.observacao += ' (em segundo plano)';
            log('   o site restaura em segundo plano: o conteúdo aparece na sala em alguns minutos', 'aviso');
          }
        } catch (e) {
          registro.estado = 'erro';
          registro.observacao = e.message;
          log(`   erro: ${e.message}`, 'erro');
          // Perder matrícula é o único erro que não pode se repetir sala a
          // sala: para tudo na hora.
          if (/^ALUNOS PERDIDOS/.test(e.message)) {
            log('PAREI TUDO. Confira esta sala antes de seguir.', 'erro');
            interromper = true;
          }
        }
        await dorme(CONFIG.pausaMs);
      }

      sobraramArquivos = [...arquivosPorChave.keys()].filter((k) => !usados.has(k));
      const sobraram = sobraramArquivos;
      if (sobraram.length) {
        log('');
        log(`${sobraram.length} arquivo(s) sem sala correspondente nesta categoria:`, 'aviso');
        sobraram.forEach((k) => log(`   ${arquivosPorChave.get(k).name}`, 'aviso'));
      }

      resumoFinal(sobraram);
      if (relatorio.length) botaoCopiar.style.display = '';
    } catch (e) {
      log(`Parou: ${e.message}`, 'erro');
    } finally {
      rodando = false; interromper = false; perguntarSempre = true;
      travaBotoes(false); botaoParar.style.display = 'none'; botaoParar.textContent = 'Parar';
    }
  }

  const ESTADOS = [
    ['restaurada', 'RESTAURADAS'],
    ['restauraria', 'SALAS PRONTAS PARA RESTAURAR (só conferido)'],
    ['sem arquivo', 'SEM ARQUIVO CORRESPONDENTE'],
    ['pulada', 'PULADAS POR VOCÊ'],
    ['erro', 'COM ERRO'],
  ];

  function textoResumo(sobraram = sobraramArquivos) {
    const linhas = [`RESTAURAÇÃO DOS BACKUPS — ${new Date().toLocaleString('pt-BR')}`];
    linhas.push(`${relatorio.length} sala(s) percorrida(s).`);

    for (const [estado, titulo] of ESTADOS) {
      const itens = relatorio.filter((r) => r.estado === estado);
      if (!itens.length) continue;
      linhas.push('');
      linhas.push(`${titulo} (${itens.length})`);
      for (const r of itens) {
        linhas.push(`  - ${r.curto}${r.observacao ? ` — ${r.observacao}` : ''}`);
      }
    }

    if (sobraram && sobraram.length) {
      linhas.push('');
      linhas.push(`ARQUIVOS SEM SALA NESTA CATEGORIA (${sobraram.length})`);
      sobraram.forEach((k) => linhas.push(`  - ${arquivosPorChave.get(k).name}`));
    }
    return linhas.join('\n');
  }

  function resumoFinal(sobraram) {
    log('');
    log('════════ RESUMO ════════');
    for (const linha of textoResumo(sobraram).split('\n')) {
      const tipo = /^RESTAURADAS/.test(linha) ? 'ok'
        : (/^(SEM ARQUIVO|COM ERRO|PULADAS|ARQUIVOS SEM)/.test(linha) ? 'aviso' : undefined);
      log(linha, tipo);
    }
  }

  function copiarResumo() {
    if (!relatorio.length) { log('Rode a restauração antes de copiar o resumo.', 'aviso'); return; }
    navigator.clipboard.writeText(textoResumo())
      .then(() => log('Resumo copiado para a área de transferência.', 'ok'))
      .catch(() => log('Não consegui copiar automaticamente — selecione o texto no painel.', 'aviso'));
  }

  // ----------------------------------------------------------------------
  // Painel
  // ----------------------------------------------------------------------
  const painel = document.createElement('div');
  painel.style.cssText = `position:fixed;top:10px;right:10px;z-index:99999;width:400px;
    font:12px/1.45 system-ui,sans-serif;background:#fff;border:1px solid #ccd;border-radius:8px;
    box-shadow:0 6px 20px rgba(0,0,0,.18);padding:10px;`;

  const titulo = document.createElement('div');
  titulo.textContent = 'Restaurar backups nas salas';
  // (o modo — apagar ou mesclar — fica em CONFIG.modoDeRestauracao)
  titulo.style.cssText = 'font-weight:600;margin-bottom:8px;';

  const entrada = document.createElement('input');
  entrada.type = 'file';
  entrada.multiple = true;
  entrada.accept = '.mbz';
  entrada.style.cssText = 'display:block;margin-bottom:8px;width:100%;';
  entrada.onchange = () => registraArquivos([...entrada.files]);

  const botoes = document.createElement('div');
  botoes.style.cssText = 'display:flex;flex-wrap:wrap;gap:6px;';

  const criaBotao = (texto, cor, acao) => {
    const b = document.createElement('button');
    b.textContent = texto;
    b.style.cssText = `padding:8px 10px;background:${cor};color:#fff;border:0;border-radius:5px;cursor:pointer;`;
    b.onclick = acao;
    botoes.appendChild(b);
    return b;
  };

  const botao = criaBotao('Restaurar', '#28a745', () => executar(false));
  const botaoConferir = criaBotao('Só conferir', '#5a6673', () => executar(true));
  const botaoDiag = criaBotao('Diagnosticar', '#2d5bd0', diagnosticar);
  const botaoCopiar = criaBotao('Copiar resumo', '#5b21b6', copiarResumo);
  botaoCopiar.style.display = 'none';
  const botaoParar = criaBotao('Parar', '#b3261e', () => { interromper = true; botaoParar.textContent = 'Parando…'; });
  botaoParar.style.display = 'none';

  const travaBotoes = (t) => {
    botao.disabled = botaoConferir.disabled = botaoDiag.disabled = t;
    botaoCopiar.disabled = t;
  };

  const painelLog = document.createElement('div');
  painelLog.style.cssText = 'margin-top:8px;max-height:340px;overflow:auto;white-space:pre-wrap;font-family:ui-monospace,monospace;font-size:11px;';

  painel.append(titulo, entrada, botoes, painelLog);
  document.body.appendChild(painel);

  function log(msg, tipo) {
    const cor = { ok: '#177245', erro: '#b3261e', aviso: '#8a6100' }[tipo] || '#333';
    const linha = document.createElement('div');
    linha.style.color = cor;
    linha.style.fontWeight = tipo ? '600' : 'normal';
    linha.textContent = msg;
    painelLog.appendChild(linha);
    painelLog.scrollTop = painelLog.scrollHeight;
    console.log('[UGB restaurar]', msg);
  }
